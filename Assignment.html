%python # ================================================================ # BCS GAP DETECTION ENGINE — 9-COLUMN OUTPUT # ================================================================ # Output columns: # gap_id → unique key # member_id → member identifier # gap_status → OPEN / CLOSED / EXCLUDED # gap_description → human readable reason # in_denominator → eligible for BCS measure # is_excluded → removed from denominator # numerator_met → mammogram found in window # exclusion_reason → Bilateral Mastectomy / Hospice # numerator_evidence_code→ CPT or HCPCS code that closed gap # # All logic is reference-table driven — no hardcoded codes # ================================================================ # ================================================================ # CELL 1 — CONFIGURATION # ================================================================ YEAR = 2025 START = "2025-01-01" END = "2025-12-31" RUN_ID = f"BCS_RUN_{YEAR}_001" CATALOG = "aiagenticdemo" SILVER = f"{CATALOG}.hedisabstractor" GOLD = f"{CATALOG}.hedisabstractor" REF = f"{CATALOG}.hedisabstractor" BCS_LOOKBACK_MONTHS = 27 spark.sql(f"USE CATALOG {CATALOG}") print("=" * 55) print("BCS GAP DETECTION ENGINE — 9-COLUMN OUTPUT") print("=" * 55) print(f" Catalog : {CATALOG}") print(f" Silver : {SILVER}") print(f" Gold out : {GOLD}.bcs_gap") print(f" Year : {YEAR}") print(f" Lookback : {BCS_LOOKBACK_MONTHS} months") print("=" * 55) # ================================================================ # CELL 2 — VERIFY REFERENCE TABLE # ================================================================ print("\nVerifying reference table...") ref_count = spark.sql(f""" SELECT COUNT(*) AS n FROM {REF}.hedis_value_sets WHERE measure_code = 'BCS' AND is_active = TRUE AND effective_year = {YEAR} """).collect()[0]["n"] print(f" ✅ {ref_count} BCS codes in reference table") spark.sql(f""" SELECT measure_component, code_system, code, code_description FROM {REF}.hedis_value_sets WHERE measure_code = 'BCS' AND is_active = TRUE AND effective_year = {YEAR} ORDER BY measure_component, code_system, code """).show(50, truncate=False) # ================================================================ # CELL 3 — EXCLUSION A: HOSPICE # Covers: HCPCS Q codes + POS 34 + ICD Z51.5 # All codes pulled from reference.hedis_value_sets # ================================================================ print("\n" + "=" * 55) print("STEP 1: Hospice exclusion...") excl_hospice_df = spark.sql(f""" -- HCPCS hospice codes from reference table SELECT DISTINCT cp.member_id FROM {SILVER}.claims_procedure cp INNER JOIN {REF}.hedis_value_sets ref ON cp.procedure_code = ref.code AND ref.measure_code = 'BCS' AND ref.measure_component = 'Exclusion' AND ref.code_system = 'HCPCS' AND ref.is_active = TRUE AND ref.effective_year = {YEAR} WHERE YEAR(cp.service_date) = {YEAR} UNION -- Place of service 34 always means hospice SELECT DISTINCT member_id FROM {SILVER}.claims_procedure WHERE place_of_service_code = '34' AND YEAR(service_date) = {YEAR} UNION -- ICD-10 hospice diagnosis codes (Z51.5) from reference table SELECT DISTINCT cd.member_id FROM {SILVER}.claims_diagnoses cd INNER JOIN {REF}.hedis_value_sets ref ON cd.diagnosis_code = ref.code AND ref.measure_code = 'BCS' AND ref.measure_component = 'Exclusion' AND ref.code_system = 'ICD10CM' AND ref.is_active = TRUE AND ref.effective_year = {YEAR} WHERE YEAR(cd.service_date) = {YEAR} """) excl_hospice_df.createOrReplaceTempView("excl_hospice_bcs") print(f" ✅ Hospice exclusion — {excl_hospice_df.select('member_id').distinct().count()} members") # ================================================================ # CELL 4 — EXCLUSION B: BILATERAL MASTECTOMY # Covers 3 scenarios — all code lookups via reference table: # 1. Z90.13 or Z90.10 — direct bilateral codes # 2. Z90.11 (right) + Z90.12 (left) both present = bilateral # 3. CPT mastectomy code with bilateral modifier/flag # # NOT excluded: Z90.12 alone (left only) — e.g. M007 scenario # ================================================================ print("\n" + "=" * 55) print("STEP 2: Bilateral mastectomy exclusion...") excl_bilateral_df = spark.sql(f""" -- SCENARIO 1: Z90.13 or Z90.10 — direct bilateral ICD codes SELECT DISTINCT cd.member_id FROM {SILVER}.claims_diagnoses cd INNER JOIN {REF}.hedis_value_sets ref ON cd.diagnosis_code = ref.code AND ref.measure_code = 'BCS' AND ref.measure_component = 'Exclusion' AND ref.code_system = 'ICD10CM' AND ref.code IN ('Z90.13', 'Z90.10') AND ref.is_active = TRUE AND ref.effective_year = {YEAR} UNION -- SCENARIO 2: Both Z90.11 (right) AND Z90.12 (left) present -- M007 has only left (Z90.12) → will NOT appear here → correctly not excluded SELECT DISTINCT right_side.member_id FROM ( SELECT DISTINCT cd.member_id FROM {SILVER}.claims_diagnoses cd INNER JOIN {REF}.hedis_value_sets ref ON cd.diagnosis_code = ref.code AND ref.measure_code = 'BCS' AND ref.code = 'Z90.11' AND ref.is_active = TRUE ) right_side INNER JOIN ( SELECT DISTINCT cd.member_id FROM {SILVER}.claims_diagnoses cd INNER JOIN {REF}.hedis_value_sets ref ON cd.diagnosis_code = ref.code AND ref.measure_code = 'BCS' AND ref.code = 'Z90.12' AND ref.is_active = TRUE ) left_side ON right_side.member_id = left_side.member_id UNION -- SCENARIO 3: Mastectomy CPT with bilateral modifier or flag SELECT DISTINCT cp.member_id FROM {SILVER}.claims_procedure cp INNER JOIN {REF}.hedis_value_sets ref ON cp.procedure_code = ref.code AND ref.measure_code = 'BCS' AND ref.measure_component = 'Exclusion' AND ref.code_system = 'CPT' AND ref.is_active = TRUE AND ref.effective_year = {YEAR} WHERE cp.modifier_1 = '50' OR cp.modifier_2 = '50' OR cp.laterality = 'Bilateral' OR cp.is_bcs_bilateral_mastectomy = TRUE """) excl_bilateral_df.createOrReplaceTempView("excl_bilateral_mastectomy") bilateral_count = excl_bilateral_df.select("member_id").distinct().count() print(f" ✅ Bilateral mastectomy exclusion — {bilateral_count} members") # Verify M007 is NOT excluded (left mastectomy only) m007 = excl_bilateral_df.filter("member_id = 'M007'").count() print(f" ✅ M007 in exclusion: {m007} rows (expected 0 — left only, not bilateral)") # ================================================================ # CELL 5 — DENOMINATOR # Criteria: F, age 52-74, enrolled, Commercial/Medicare, medical benefit # ================================================================ print("\n" + "=" * 55) print("STEP 3: Building denominator...") denom_df = spark.sql(f""" SELECT DISTINCT m.member_id FROM {SILVER}.member_demographics m JOIN {SILVER}.member_enrollment e ON m.member_id = e.member_id WHERE m.gender = 'F' AND m.age_as_of_year_end BETWEEN 52 AND 74 AND m.member_status = 'Active' AND m.deceased_date IS NULL AND e.measurement_year = {YEAR} AND ( e.continuous_enrollment_my2025 = TRUE OR (e.has_allowable_gap = TRUE AND e.enrollment_gap_days <= 45) ) AND e.product_line IN ('Commercial', 'Medicare') AND e.has_medical_benefit = TRUE AND e.eligible_bcs = TRUE """) denom_df.createOrReplaceTempView("denom_bcs") print(f" ✅ Denominator — {denom_df.count()} eligible members") denom_df.show(truncate=False) # ================================================================ # CELL 6 — NUMERATOR # Evidence: mammogram CPT/HCPCS/UBREV from reference table # Rules: Paid, in-person, within 27-month window, within enrollment # ================================================================ print("\n" + "=" * 55) print("STEP 4: Building numerator (most recent mammogram)...") numer_df = spark.sql(f""" WITH mammogram_claims AS ( SELECT cp.member_id, cp.procedure_code, ref.code AS matched_code, ref.code_system AS matched_code_system, ROW_NUMBER() OVER ( PARTITION BY cp.member_id ORDER BY cp.service_date DESC, cp.claim_id DESC ) AS rn FROM {SILVER}.claims_procedure cp -- Match CPT and HCPCS codes to reference numerator codes INNER JOIN {REF}.hedis_value_sets ref ON ( (cp.procedure_code = ref.code AND ref.code_system IN ('CPT','HCPCS')) OR (cp.revenue_code = ref.code AND ref.code_system = 'UBREV') ) AND ref.measure_code = 'BCS' AND ref.measure_component = 'Numerator' AND ref.is_active = TRUE AND ref.effective_year = {YEAR} -- Validate mammogram falls within member's enrollment period INNER JOIN {SILVER}.member_enrollment e ON cp.member_id = e.member_id AND e.measurement_year = {YEAR} AND cp.service_date BETWEEN e.enrollment_start_date AND COALESCE(e.enrollment_end_date, TO_DATE('{END}')) WHERE cp.claim_status = 'Paid' AND cp.service_date >= ADD_MONTHS(TO_DATE('{END}'), -{BCS_LOOKBACK_MONTHS}) AND cp.service_date <= TO_DATE('{END}') AND COALESCE(cp.place_of_service_code, '11') NOT IN ('02', '10') ) SELECT member_id, matched_code, matched_code_system FROM mammogram_claims WHERE rn = 1 """) numer_df.createOrReplaceTempView("numer_bcs") print(f" ✅ Numerator — {numer_df.count()} members with mammogram evidence") numer_df.show(truncate=False) # ================================================================ # CELL 7 — GAP DETECTION LOGIC → 9 COLUMNS ONLY # gap_id, member_id, gap_status, gap_description, # in_denominator, is_excluded, numerator_met, # exclusion_reason, numerator_evidence_code # ================================================================ print("\n" + "=" * 55) print("STEP 5: Running gap detection → 9 columns...") gaps_df = spark.sql(f""" SELECT -- ── 1. gap_id ────────────────────────────────────────── CONCAT(d.member_id, '_BCS_{YEAR}') AS gap_id, -- ── 2. member_id ─────────────────────────────────────── d.member_id, -- ── 3. gap_status ────────────────────────────────────── -- Rule: exclusion wins → then closed → else open CASE WHEN bm.member_id IS NOT NULL THEN 'EXCLUDED' WHEN h.member_id IS NOT NULL THEN 'EXCLUDED' WHEN n.member_id IS NOT NULL THEN 'CLOSED' ELSE 'OPEN' END AS gap_status, -- ── 4. gap_description ───────────────────────────────── CASE WHEN bm.member_id IS NOT NULL THEN 'EXCLUDED: Bilateral mastectomy on record — BCS not applicable' WHEN h.member_id IS NOT NULL THEN 'EXCLUDED: Member received hospice care in 2025' WHEN n.member_id IS NOT NULL THEN CONCAT('CLOSED: Mammogram found — CPT/HCPCS code ', n.matched_code, ' (', n.matched_code_system, ')') ELSE CONCAT('OPEN: No mammogram found in 27-month window ', CAST(ADD_MONTHS(TO_DATE('{END}'), -{BCS_LOOKBACK_MONTHS}) AS STRING), ' to {END}') END AS gap_description, -- ── 5. in_denominator ────────────────────────────────── -- TRUE when member qualifies AND is not excluded CASE WHEN bm.member_id IS NULL AND h.member_id IS NULL THEN TRUE ELSE FALSE END AS in_denominator, -- ── 6. is_excluded ───────────────────────────────────── CASE WHEN bm.member_id IS NOT NULL OR h.member_id IS NOT NULL THEN TRUE ELSE FALSE END AS is_excluded, -- ── 7. numerator_met ─────────────────────────────────── CASE WHEN n.member_id IS NOT NULL THEN TRUE ELSE FALSE END AS numerator_met, -- ── 8. exclusion_reason ──────────────────────────────── CASE WHEN bm.member_id IS NOT NULL THEN 'Bilateral Mastectomy' WHEN h.member_id IS NOT NULL THEN 'Hospice' ELSE NULL END AS exclusion_reason, -- ── 9. numerator_evidence_code ───────────────────────── -- The CPT or HCPCS code that closed the gap -- NULL when gap is OPEN or EXCLUDED n.matched_code AS numerator_evidence_code FROM denom_bcs d LEFT JOIN excl_bilateral_mastectomy bm ON d.member_id = bm.member_id LEFT JOIN excl_hospice_bcs h ON d.member_id = h.member_id LEFT JOIN numer_bcs n ON d.member_id = n.member_id """) gaps_df.createOrReplaceTempView("bcs_gap_final") total = gaps_df.count() open_cnt = gaps_df.filter("gap_status = 'OPEN'").count() closed_cnt = gaps_df.filter("gap_status = 'CLOSED'").count() excluded_cnt = gaps_df.filter("gap_status = 'EXCLUDED'").count() print(f"\n ✅ Gap detection complete") print(f" TOTAL : {total}") print(f" OPEN : {open_cnt} (need outreach)") print(f" CLOSED : {closed_cnt} (mammogram found)") print(f" EXCLUDED : {excluded_cnt} (mastectomy or hospice)") print("\n All 9 columns:") gaps_df.select( "gap_id", "member_id", "gap_status", "in_denominator", "is_excluded", "numerator_met", "exclusion_reason", "numerator_evidence_code", "gap_description" ).orderBy("gap_status", "member_id").show(50, truncate=False) # ================================================================ # CELL 8 — WRITE TO bcs_gap TABLE # ================================================================ print("\n" + "=" * 55) print("STEP 6: Writing to bcs_gap table...") # Create table with exactly 9 columns spark.sql(f""" CREATE TABLE IF NOT EXISTS {GOLD}.bcs_gap ( gap_id STRING COMMENT 'Unique key: member_id + BCS + year', member_id STRING COMMENT 'Member identifier — FK to silver tables', gap_status STRING COMMENT 'OPEN / CLOSED / EXCLUDED', gap_description STRING COMMENT 'Human readable reason for gap status', in_denominator BOOLEAN COMMENT 'TRUE if member qualifies for BCS and is not excluded', is_excluded BOOLEAN COMMENT 'TRUE if bilateral mastectomy or hospice', numerator_met BOOLEAN COMMENT 'TRUE if mammogram found in 27-month window', exclusion_reason STRING COMMENT 'Bilateral Mastectomy or Hospice — NULL if not excluded', numerator_evidence_code STRING COMMENT 'CPT/HCPCS code that closed the gap — NULL if OPEN or EXCLUDED' ) USING DELTA COMMENT 'BCS gap detection — 9 core columns — HEDIS 2025' """) # Safe re-run: delete before insert spark.sql(f"DELETE FROM {GOLD}.bcs_gap WHERE gap_id LIKE '%_BCS_{YEAR}'") # Write 9-column result gaps_df.select( "gap_id", "member_id", "gap_status", "gap_description", "in_denominator", "is_excluded", "numerator_met", "exclusion_reason", "numerator_evidence_code" ).write \ .format("delta") \ .mode("append") \ .saveAsTable(f"{GOLD}.bcs_gap") written = spark.sql(f""" SELECT COUNT(*) AS n FROM {GOLD}.bcs_gap WHERE gap_id LIKE '%_BCS_{YEAR}' """).collect()[0]["n"] print(f" ✅ Written {written} rows to {GOLD}.bcs_gap") # ================================================================ # CELL 9 — VERIFY RESULTS # ================================================================ print("\n" + "=" * 55) print("STEP 7: Verification...") print("=" * 55) # V1: Summary print("\n📊 V1 — Gap summary:") spark.sql(f""" SELECT gap_status, COUNT(*) AS members, COUNT(CASE WHEN numerator_evidence_code IS NOT NULL THEN 1 END) AS has_evidence_code, COUNT(CASE WHEN exclusion_reason IS NOT NULL THEN 1 END) AS has_exclusion_reason FROM {GOLD}.bcs_gap GROUP BY gap_status ORDER BY gap_status """).show() # V2: HEDIS compliance rate print("\n📊 V2 — HEDIS BCS compliance rate:") spark.sql(f""" SELECT COUNT(CASE WHEN in_denominator = TRUE THEN 1 END) AS denominator, COUNT(CASE WHEN is_excluded = TRUE THEN 1 END) AS excluded, COUNT(CASE WHEN numerator_met = TRUE THEN 1 END) AS numerator, COUNT(CASE WHEN gap_status = 'OPEN' THEN 1 END) AS open_gaps, ROUND( COUNT(CASE WHEN numerator_met = TRUE THEN 1 END) * 100.0 / NULLIF(COUNT(CASE WHEN in_denominator = TRUE THEN 1 END), 0) , 1) AS compliance_pct FROM {GOLD}.bcs_gap """).show() # V3: Open gaps print("\n📊 V3 — Open gaps (need outreach):") spark.sql(f""" SELECT gap_id, member_id, gap_status, exclusion_reason, numerator_evidence_code, gap_description FROM {GOLD}.bcs_gap WHERE gap_status = 'OPEN' ORDER BY member_id """).show(truncate=False) # V4: Excluded members print("\n📊 V4 — Excluded members:") spark.sql(f""" SELECT gap_id, member_id, gap_status, exclusion_reason, gap_description FROM {GOLD}.bcs_gap WHERE gap_status = 'EXCLUDED' ORDER BY exclusion_reason, member_id """).show(truncate=False) # V5: Closed gaps print("\n📊 V5 — Closed gaps (mammogram found):") spark.sql(f""" SELECT gap_id, member_id, gap_status, numerator_evidence_code, gap_description FROM {GOLD}.bcs_gap WHERE gap_status = 'CLOSED' ORDER BY member_id """).show(truncate=False) # V6: Validate M007 — unilateral only → should be OPEN not EXCLUDED print("\n📊 V6 — M007 check (left mastectomy only — must be OPEN):") spark.sql(f""" SELECT gap_id, member_id, gap_status, is_excluded, exclusion_reason, gap_description FROM {GOLD}.bcs_gap WHERE member_id = 'M007' """).show(truncate=False) # V7: Validate M004 — bilateral → should be EXCLUDED print("\n📊 V7 — M004 check (bilateral mastectomy — must be EXCLUDED):") spark.sql(f""" SELECT gap_id, member_id, gap_status, is_excluded, exclusion_reason, gap_description FROM {GOLD}.bcs_gap WHERE member_id = 'M004' """).show(truncate=False) # V8: All 9 columns schema confirmation print("\n📊 V8 — Table schema (confirm 9 columns):") spark.sql(f"DESCRIBE TABLE {GOLD}.bcs_gap").show(truncate=False) # ================================================================ # CELL 10 — FINAL SUMMARY # ================================================================ print("\n" + "=" * 55) print("BCS GAP DETECTION — COMPLETE") print("=" * 55) row = spark.sql(f""" SELECT COUNT(*) AS total, COUNT(CASE WHEN gap_status='OPEN' THEN 1 END) AS open_cnt, COUNT(CASE WHEN gap_status='CLOSED' THEN 1 END) AS closed_cnt, COUNT(CASE WHEN gap_status='EXCLUDED' THEN 1 END) AS excl_cnt, ROUND( COUNT(CASE WHEN numerator_met = TRUE THEN 1 END) * 100.0 / NULLIF(COUNT(CASE WHEN in_denominator = TRUE THEN 1 END), 0) , 1) AS rate FROM {GOLD}.bcs_gap """).collect()[0] print(f" Output table : {GOLD}.bcs_gap") print(f" Columns : 9 (gap_id, member_id, gap_status, gap_description,") print(f" in_denominator, is_excluded, numerator_met,") print(f" exclusion_reason, numerator_evidence_code)") print(f" Total rows : {row['total']}") print(f" OPEN : {row['open_cnt']}") print(f" CLOSED : {row['closed_cnt']}") print(f" EXCLUDED : {row['excl_cnt']}") print(f" Compliance : {row['rate']}%") print("=" * 55) print(" All codes sourced from reference.hedis_value_sets") print(" No hardcoded CPT/HCPCS/ICD codes in logic") print("=" * 55)
