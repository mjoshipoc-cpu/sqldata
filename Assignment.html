SELECT WAREHOUSE_NAME, SUM(CREDITS_USED) AS total_credits, SUM(CREDITS_USED_COMPUTE) AS compute_credits, SUM(CREDITS_USED_CLOUD_SERVICES) AS cloud_credits, ROUND(SUM(CREDITS_USED) * 3, 2) AS est_cost_usd -- approx $3/credit FROM SNOWFLAKE.ACCOUNT_USAGE.WAREHOUSE_METERING_HISTORY WHERE START_TIME >= DATE_TRUNC('MONTH', CURRENT_DATE) GROUP BY WAREHOUSE_NAME ORDER BY total_credits DESC;
