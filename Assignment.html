================================================================ # CELL 3 — VERIFY SILVER TABLES # ================================================================ print("\nVerifying silver tables...") for tbl in ["member_demographics","member_enrollment", "claims_diagnoses","lab_results","claims_procedure"]: try: n = spark.sql(f"SELECT COUNT(*) AS n FROM {SILVER}.{tbl}").collect()[0]["n"] print(f" ✅ {SILVER}.{tbl} — {n} rows") except Exception as e: print(f" ❌ {SILVER}.{tbl} — {e}") # ================================================================ # CELL 4 — EXCLUSION A: ESRD / DIALYSIS # Codes : N18.5, N18.6, Z99.2, Z91.15 # Rule : Historical — any date in patient history # ================================================================ print("\n" + "=" * 55) print("STEP 1a: ESRD/Dialysis exclusion...") excl_esrd_df = spark.sql(f""" SELECT DISTINCT cd.member_id, ref.code AS exclusion_code, 'ESRD_Dialysis' AS exclusion_type FROM {SILVER}.claims_diagnoses cd INNER JOIN {REF}.hedis_value_sets ref ON cd.diagnosis_code = ref.code AND ref.measure_code = 'CBP' AND ref.measure_component = 'Exclusion' AND ref.code_system = 'ICD10CM' -- Codes come from reference table — no hardcoding needed AND ref.is_active = TRUE AND ref.effective_year = {YEAR} """) excl_esrd_df.createOrReplaceTempView("excl_esrd_cbp") print(f" ✅ ESRD — {excl_esrd_df.select('member_id').distinct().count()} members") # ================================================================ print("\n" + "=" * 55) print(f"STEP 1c: Frailty exclusion (age {FRAILTY_AGE}+ only)...") excl_frailty_df = spark.sql(f""" SELECT DISTINCT cd.member_id, ref.code AS exclusion_code, 'Frailty_Advanced_Illness' AS exclusion_type FROM {SILVER}.claims_diagnoses cd INNER JOIN {SILVER}.member_demographics m ON cd.member_id = m.member_id AND m.age_as_of_year_end >= {FRAILTY_AGE} INNER JOIN {REF}.hedis_value_sets ref ON cd.diagnosis_code = ref.code AND ref.measure_code = 'CBP' AND ref.measure_component = 'Exclusion' AND ref.code_system = 'ICD10CM' -- Codes come from reference table — no hardcoding needed AND ref.is_active = TRUE AND ref.effective_year = {YEAR} WHERE YEAR(cd.service_date) = {YEAR} """) excl_frailty_df.createOrReplaceTempView("excl_frailty_cbp") print(f" ✅ Frailty (age 66+) — {excl_frailty_df.select('member_id').distinct().count()} members") # ================================================================ # CELL 8 — DENOMINATOR # Criteria: # 1. ================================================================ print("\n" + "=" * 55) print("STEP 2: Building CBP denominator...") denom_df = spark.sql(f""" SELECT DISTINCT m.member_id, ref.measure_code = 'CBP' AND ref.measure_component = 'Denominator' AND ref.code_system = 'ICD10CM' AND ref.is_active = TRUE AND ref.effective_year = {YEAR} WHERE YEAR(cd.service_date) = {YEAR} ) htn_dx ON m.member_id = htn_dx.member_id WHERE m.age_as_of_year_end BETWEEN 18 AND 85 AND m.member_status = 'Active' AND m.deceased_date IS NULL AND e.measurement_year = {YEAR} AND ( e.continuous_enrollment_my2025 = TRUE OR (e.has_allowable_gap = TRUE AND e.enrollment_gap_days <= 45) ) AND e.product_line IS NOT NULL AND e.has_medical_benefit = TRUE AND e.eligible_cbp = TRUE """) denom_df.createOrReplaceTempView("denom_cbp") print(f" ✅ Denominator — {denom_df.count()} members") denom_df.show(truncate=False) # ================================================================ # CELL 9 — NUMERATOR # Rule : MOST RECENT BP reading in measurement year # Controlled = systolic < 140 AND diastolic < 90 # Source: lab_results via LOINC codes from reference table # + encounter flag for readings without LOINC # # Returns ALL readings (controlled and uncontrolled) # so gap logic can set sub_type correctly # ================================================================ print("\n" + "=" * 55) print("STEP 3: Building CBP numerator (most recent BP reading)...") numer_df = spark.sql(f""" WITH all_bp AS ( -- Source 1: LOINC-coded BP readings from reference table SELECT lr.member_id, lr.systolic_bp, lr.diastolic_bp, lr.lab_test_date AS bp_date, ref.code AS loinc_code, CONCAT( CAST(lr.systolic_bp AS STRING), '/', CAST(lr.diastolic_bp AS STRING) ) AS bp_value, CASE WHEN lr.systolic_bp < {SYSTOLIC_THRESHOLD} AND lr.diastolic_bp < {DIASTOLIC_THRESHOLD} THEN TRUE ELSE FALSE END AS bp_controlled FROM {SILVER}.lab_results lr INNER JOIN {REF}.hedis_value_sets ref ON lr.lab_test_code = ref.code AND ref.measure_code = 'CBP' AND ref.measure_component = 'Numerator' AND ref.code_system = 'LOINC' AND ref.is_active = TRUE AND ref.effective_year = {YEAR} WHERE lr.lab_test_date BETWEEN '{START}' AND '{END}' AND lr.systolic_bp IS NOT NULL AND lr.diastolic_bp IS NOT NULL AND lr.systolic_bp > 0 AND lr.diastolic_bp > 0 UNION ALL -- Source 2: BP readings via encounter flag (no LOINC match needed) SELECT lr.member_id, lr.systolic_bp, lr.diastolic_bp, lr.lab_test_date AS bp_date, 'CBP_Encounter' AS loinc_code, CONCAT( CAST(lr.systolic_bp AS STRING), '/', CAST(lr.diastolic_bp AS STRING) ) AS bp_value, CASE WHEN lr.systolic_bp < {SYSTOLIC_THRESHOLD} AND lr.diastolic_bp < {DIASTOLIC_THRESHOLD} THEN TRUE ELSE FALSE END AS bp_controlled FROM {SILVER}.lab_results lr WHERE lr.is_cbp_bp_reading = TRUE AND lr.lab_test_date BETWEEN '{START}' AND '{END}' AND lr.systolic_bp IS NOT NULL AND lr.diastolic_bp IS NOT NULL AND lr.systolic_bp > 0 AND lr.diastolic_bp > 0 ), -- Pick MOST RECENT reading per member -- NCQA rule: only the latest reading determines gap status ranked AS ( SELECT *, ROW_NUMBER() OVER ( PARTITION BY member_id ORDER BY bp_date DESC ) AS rn FROM all_bp ) SELECT member_id, systolic_bp, diastolic_bp, bp_date, loinc_code, bp_value, bp_controlled FROM ranked WHERE rn = 1 """) numer_df.createOrReplaceTempView("numer_cbp") total_bp = numer_df.count() controlled = numer_df.filter("bp_controlled = TRUE").count() uncontrolled = numer_df.filter("bp_controlled = FALSE").count() print(f" ✅ BP readings found : {total_bp}") print(f" Controlled (<140/90) : {controlled} → CLOSED") print(f" Uncontrolled (>=140/90): {uncontrolled} → OPEN/UNCONTROLLED") print(f" No reading : {denom_df.count() - total_bp} → OPEN/NO_READING") numer_df.show(truncate=False) # ================================================================ # CELL 10 — GAP DETECTION → EXACTLY 14 COLUMNS # ================================================================ print("\n" + "=" * 55) print("STEP 4: Gap detection → 14 columns...") gaps_df = spark.sql(f""" SELECT -- ── Column 1: gap_id care_gaps table {YEAR} AS measurement_year, -- ── Column 5: gap_status ────────────────────────────── -- Priority: exclusion wins → closed → open CASE WHEN esrd.member_id IS NOT NULL THEN 'EXCLUDED' WHEN preg.member_id IS NOT NULL THEN 'EXCLUDED' WHEN frail.member_id IS NOT NULL THEN 'EXCLUDED' WHEN hosp.member_id IS NOT NULL THEN 'EXCLUDED' WHEN n.member_id IS NOT NULL AND n.bp_controlled = TRUE THEN 'CLOSED' ELSE 'OPEN' END AS gap_status, -- ── Column 6: gap_sub_type ──────────────────────────── -- CBP-specific — NULL in bcs_gap rows -- Tells coordinator WHY gap is open so they say the right thing CASE WHEN esrd.member_id IS NOT NULL THEN NULL WHEN preg.member_id IS NOT NULL THEN NULL WHEN frail.member_id IS NOT NULL THEN NULL WHEN hosp.member_id IS NOT NULL THEN NULL WHEN n.member_id IS NOT NULL AND n.bp_controlled = FALSE THEN 'UNCONTROLLED' WHEN n.member_id IS NULL THEN 'NO_READING' ELSE NULL END AS gap_sub_type, -- ── Column 7: gap_description ───────────────────────── CASE WHEN esrd.member_id IS NOT NULL THEN CONCAT('EXCLUDED: ESRD/Dialysis — ICD ', esrd.exclusion_code) WHEN preg.member_id IS NOT NULL THEN CONCAT('EXCLUDED: Pregnancy HTN — ICD ', preg.exclusion_code) WHEN frail.member_id IS NOT NULL THEN CONCAT('EXCLUDED: Frailty (age 66+) — ICD ', frail.exclusion_code) WHEN hosp.member_id IS NOT NULL THEN CONCAT('EXCLUDED: Hospice — code ', hosp.exclusion_code) WHEN n.bp_controlled = FALSE THEN CONCAT('OPEN: Most recent BP ', n.bp_value, ' on ', CAST(n.bp_date AS STRING), ' — above target 140/90') ELSE 'OPEN: No blood pressure reading found in 2025' END AS frail.member_id IS NOT NULL THEN 'Frailty (age 66+)' WHEN hosp.member_id IS NOT NULL THEN 'Hospice' ELSE NULL END AS exclusion_reason, -- ── Column 12: numerator_evidence_code ──────────────── -- excl_hospice_cbp hosp ON d.member_id = hosp.member_id LEFT JOIN numer_cbp n ON d.member_id = n.member_id """) gaps_df.createOrReplaceTempView("cbp_gap_final") total = gaps_df.count() open_cnt = OPEN : {open_cnt}") print(f" UNCONTROLLED : {uncontrolled} (BP found but >= 140/90)") print(f" NO_READING : {no_reading} (no BP in 2025)") print(f" CLOSED : {closed_cnt} (BP < 140/90)") print(f" EXCLUDED : {excluded_cnt}") print("\n All 14 columns:") gaps_df.select( "gap_id","member_id","measure_code","measurement_year", "gap_status","gap_sub_type","in_denominator","is_excluded",
