import jwt import json from utils.logutils import logger from utils.confighelper import ConfigHelper from utils.db_utils import query_configdb class Jwtutils: def __init__(self, clientname=None): self.logger_obj = logger() self.clientname = clientname ch = ConfigHelper(clientname=self.clientname) keys = ch.load_keys() self.client_private_key = keys[0] self.client_public_key = keys[1] self.exl_private_key = keys[2] self.exl_public_key = keys[3] self.jwtconfig = ch.getJWTSettings() def decode_jwt(self, token, internal = True): try: payload = jwt.decode( token, self.exl_public_key if internal else self.client_public_key, algorithms=[f"{self.jwtconfig['jwt_algorithm']}"], audience="interop.exlservice.com" ) return payload except jwt.ExpiredSignatureError: self.logger_obj.log("Token expired") raise ValueError("Token expired") except jwt.InvalidTokenError: self.logger_obj.log("Invalid token") raise ValueError("Invalid token") def encode_jwt(self, payload): return jwt.encode( payload, self.exl_private_key, algorithm=f"{self.jwtconfig['jwt_algorithm']}" ) def validate_jwt_sub(self, payload, expected_sub): if payload.get('sub') != expected_sub or payload.get('aud') != "interop.exlservice.com": self.logger_obj.log("sub claim does not match") raise ValueError("sub claim does not match") def validate_jwt_claims(self, payload: dict): """ Validates JWT claims for 'sub', 'iss', and 'aud' against expected values. Returns (is_valid, mismatches) where mismatches is a list of claim names that do not match. Comparison is case-insensitive. Args: payload (dict): Decoded JWT payload. clientname (str): Expected client name for 'sub'. Returns: tuple: (bool, list) where bool is True if all claims match, False otherwise. list contains names of mismatched claims. """ expected_iss = 'interop.exlservice.com' expected_aud = 'interop.exlservice.com' mismatches = [] if str(payload.get('sub', '')).lower() != str(self.clientname).lower(): mismatches.append('sub') if str(payload.get('iss', '')).lower() != expected_iss.lower(): mismatches.append('iss') if str(payload.get('aud', '')).lower() != expected_aud.lower(): mismatches.append('aud') return len(mismatches) == 0, mismatches def get_oauth_config(self): """ Fetch OAuth configuration for a client from config.vw_clientmessage. Expected columns in view: oauthflag, oauth_api, oauth_headers, oauth_data. Returns a dict with safe defaults. """ query = """ SELECT oauthflag, oauth_api, oauth_headers, oauth_data FROM config.vw_clientmessage WHERE lower(client) = %s AND endpointdesttype = 'RESTAPI' LIMIT 1 """ defaults = { "oauthflag": False, "oauth_api": None, "oauth_headers": {}, "oauth_data": {} } if not self.clientname: return defaults try: row = query_configdb(query, (self.clientname.lower(),)) if not row: return defaults oauthflag = row.get("oauthflag", False) if isinstance(oauthflag, str): oauthflag = oauthflag.strip().lower() in {"1", "true", "t", "yes", "y"} oauth_headers = row.get("oauth_headers") or {} oauth_data = row.get("oauth_data") or {} if isinstance(oauth_headers, str): oauth_headers = json.loads(oauth_headers) if isinstance(oauth_data, str): oauth_data = json.loads(oauth_data) return { "oauthflag": bool(oauthflag), "oauth_api": row.get("oauth_api"), "oauth_headers": oauth_headers if isinstance(oauth_headers, dict) else {}, "oauth_data": oauth_data if isinstance(oauth_data, dict) else {} } except Exception as e: self.logger_obj.log(f"Failed to fetch oauth config: {str(e)}", "warning") return defaults
