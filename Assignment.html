CREATE OR REPLACE VIEW v_network_health_summary AS SELECT Year, -- Overall quality AVG(Composite_VBC_Score) AS Network_Composite_Score, AVG(Chronic_Score) AS Network_Chronic_Score, AVG(Preventive_Score) AS Network_Preventive_Score, AVG(Care_Coordination_Score) AS Network_Coordination_Score, AVG(Behavioral_Score) AS Network_Behavioral_Score, -- Provider tier distribution COUNT(DISTINCT Provider_ID) AS Total_Providers, SUM(CASE WHEN Final_Provider_Tier = 'Gold' THEN 1 ELSE 0 END) AS Gold_Providers, SUM(CASE WHEN Final_Provider_Tier = 'Silver' THEN 1 ELSE 0 END) AS Silver_Providers, SUM(CASE WHEN Final_Provider_Tier = 'Bronze' THEN 1 ELSE 0 END) AS Bronze_Providers, -- Gaps (network averages) AVG(BPD_Closure) AS BPD_Closure_Avg, AVG(CBP_Closure) AS CBP_Closure_Avg, AVG(COA_Closure) AS COA_Closure_Avg, AVG(GSD_Closure) AS GSD_Closure_Avg, AVG(LSC_Closure) AS LSC_Closure_Avg, AVG(PPC_Closure) AS PPC_Closure_Avg, AVG(TRC_Closure) AS TRC_Closure_Avg, AVG(WCC_Closure) AS WCC_Closure_Avg, -- Utilization & cost AVG(ER_Visits_per_1000) AS Network_ER_1000, AVG(Readmission_Rate) AS Network_Readmission_Rate, AVG(Avg_PMPM) AS Network_PMPM, -- Simple health label CASE WHEN AVG(Composite_VBC_Score) >= 0.75 THEN 'High-performing network' WHEN AVG(Composite_VBC_Score) >= 0.60 THEN 'Moderate network – needs improvement' ELSE 'Low-performing network – high priority for intervention' END AS Network_Health_Label FROM provider_yearly_tier GROUP BY Year;
