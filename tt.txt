"""
AllergyIntolerance invariants - https://hl7.org/fhir/R4B/allergyintolerance.html#invs

  ait-1  clinicalStatus SHALL NOT be present if verificationStatus is
         entered-in-error
  ait-2  clinicalStatus SHALL be present if verificationStatus is not
         entered-in-error

Each fn(resource: dict) -> list[issue-dict]; empty list == OK.
"""
from __future__ import annotations

from fhirpipeline.operationoutcome import issue

_VER_STATUS_SYSTEM = "http://terminology.hl7.org/CodeSystem/allergyintolerance-verification"


def _has_code(codeable_concept: dict | None, system: str, code: str) -> bool:
    if not codeable_concept:
        return False
    for c in codeable_concept.get("coding") or []:
        if c.get("system") == system and c.get("code") == code:
            return True
    return False


def _verification_is_entered_in_error(resource: dict) -> bool:
    return _has_code(resource.get("verificationStatus"), _VER_STATUS_SYSTEM, "entered-in-error")


def ait_1_clinical_status_not_entered_in_error(resource: dict) -> list[dict]:
    if resource.get("clinicalStatus") is not None and _verification_is_entered_in_error(resource):
        return [issue(
            "error", "invariant",
            "ait-1: AllergyIntolerance.clinicalStatus SHALL NOT be present if "
            "verificationStatus is entered-in-error.",
            "AllergyIntolerance.clinicalStatus",
        )]
    return []


def ait_2_clinical_status_required(resource: dict) -> list[dict]:
    if resource.get("clinicalStatus") is not None:
        return []
    if _verification_is_entered_in_error(resource):
        return []
    return [issue(
        "error", "invariant",
        "ait-2: AllergyIntolerance.clinicalStatus SHALL be present if "
        "verificationStatus is not entered-in-error.",
        "AllergyIntolerance.clinicalStatus",
    )]


ALLERGYINTOLERANCE_INVARIANTS = [
    ait_1_clinical_status_not_entered_in_error,
    ait_2_clinical_status_required,
]





"""
ServiceRequest invariants - https://hl7.org/fhir/R4B/servicerequest.html#invs

  prr-1  orderDetail SHALL only be present if code is present

Each fn(resource: dict) -> list[issue-dict]; empty list == OK.
"""
from __future__ import annotations

from fhirpipeline.operationoutcome import issue


def prr_1_order_detail_needs_code(resource: dict) -> list[dict]:
    if resource.get("orderDetail") and not resource.get("code"):
        return [issue(
            "error", "invariant",
            "prr-1: ServiceRequest.orderDetail SHALL only be present if "
            "ServiceRequest.code is present.",
            "ServiceRequest.orderDetail",
        )]
    return []


SERVICEREQUEST_INVARIANTS = [
    prr_1_order_detail_needs_code,
]



"""
Condition invariants - https://hl7.org/fhir/R4B/condition.html#invs

  con-3  clinicalStatus SHALL NOT be present if verificationStatus is
         entered-in-error
  con-4  if abatement[x] is present, clinicalStatus must be one of
         resolved, remission, inactive
  con-5  clinicalStatus SHALL be present if verificationStatus is not
         entered-in-error and category is problem-list-item

Each fn(resource: dict) -> list[issue-dict]; empty list == OK.
"""
from __future__ import annotations

from fhirpipeline.operationoutcome import issue

_VER_STATUS_SYSTEM = "http://terminology.hl7.org/CodeSystem/condition-ver-status"
_CLINICAL_SYSTEM = "http://terminology.hl7.org/CodeSystem/condition-clinical"
_CATEGORY_SYSTEM = "http://terminology.hl7.org/CodeSystem/condition-category"

# Condition.abatement[x] choice element names (R4B).
_ABATEMENT_X = (
    "abatementDateTime", "abatementAge", "abatementPeriod",
    "abatementRange", "abatementString",
)


def _has_code(codeable_concept: dict | None, system: str, code: str) -> bool:
    if not codeable_concept:
        return False
    for c in codeable_concept.get("coding") or []:
        if c.get("system") == system and c.get("code") == code:
            return True
    return False


def _verification_is_entered_in_error(resource: dict) -> bool:
    return _has_code(resource.get("verificationStatus"), _VER_STATUS_SYSTEM, "entered-in-error")


def con_3_clinical_status_not_entered_in_error(resource: dict) -> list[dict]:
    if resource.get("clinicalStatus") is not None and _verification_is_entered_in_error(resource):
        return [issue(
            "error", "invariant",
            "con-3: Condition.clinicalStatus SHALL NOT be present if "
            "verificationStatus is entered-in-error.",
            "Condition.clinicalStatus",
        )]
    return []


def con_4_abatement_needs_closed_clinical_status(resource: dict) -> list[dict]:
    present_abatement = [k for k in _ABATEMENT_X if resource.get(k) is not None]
    if not present_abatement:
        return []
    clinical_status = resource.get("clinicalStatus")
    if not any(_has_code(clinical_status, _CLINICAL_SYSTEM, code)
               for code in ("resolved", "remission", "inactive")):
        return [issue(
            "error", "invariant",
            "con-4: Condition.clinicalStatus SHALL be resolved, remission, or "
            f"inactive when abatement is present (found {', '.join(present_abatement)}).",
            "Condition.abatement[x]",
        )]
    return []


def con_5_clinical_status_needed_for_problem_list(resource: dict) -> list[dict]:
    if resource.get("clinicalStatus") is not None:
        return []
    if _verification_is_entered_in_error(resource):
        return []
    for cat in resource.get("category") or []:
        if _has_code(cat, _CATEGORY_SYSTEM, "problem-list-item"):
            return [issue(
                "error", "invariant",
                "con-5: Condition.clinicalStatus SHALL be present when category "
                "is problem-list-item and verificationStatus is not entered-in-error.",
                "Condition.clinicalStatus",
            )]
    return []


CONDITION_INVARIANTS = [
    con_3_clinical_status_not_entered_in_error,
    con_4_abatement_needs_closed_clinical_status,
    con_5_clinical_status_needed_for_problem_list,
]
