from flask import Blueprint, request, jsonify
from datetime import datetime, timezone
from utils.jwt_utils import Jwtutils
from utils.db_utils import get_mongo_db_conn
from utils.trackerutils import insert_tracking_record, tracking_data_define
from utils.logutils import logger
from fhirclient.models.patient import Patient
import json
import xml.etree.ElementTree as ET
import traceback


logger_obj = logger()

patient_bp = Blueprint('Patient', __name__)

RESOURCE_TYPE = 'Patient'
COLLECTION_NAME = 'PASPatient'


def xml_element_to_dict(element):
    """
    Converts a FHIR XML element into its JSON representation.
    """
    resource_node = {}
    for child in element:
        tag = child.tag.split('}')[-1]

        if len(child):
            value = xml_element_to_dict(child)
            for attribute_name, attribute_value in child.attrib.items():
                value[attribute_name] = attribute_value
        else:
            value = child.attrib.get('value')
            if value is None:
                value = dict(child.attrib) or (child.text or '').strip()

        if tag in resource_node:
            if not isinstance(resource_node[tag], list):
                resource_node[tag] = [resource_node[tag]]
            resource_node[tag].append(value)
        else:
            resource_node[tag] = value

    return resource_node


def get_request_resource():
    """
    Reads the incoming body, converting XML to JSON when required.
    Returns the FHIR resource as dict.
    """
    content_type = (request.content_type or '').split(';')[0].split('/')[-1].lower()
    if content_type not in ['json', 'xml']:
        raise ValueError("Unsupported content type or invalid data format")

    raw_data = request.get_data()
    if not raw_data:
        raise ValueError("no data submitted in the body")

    if content_type == 'xml':
        # convert from XML to JSON as the downstream processing is JSON only
        resource_json = xml_element_to_dict(ET.fromstring(raw_data))
        resource_json['resourceType'] = RESOURCE_TYPE
        return resource_json

    return json.loads(raw_data)


def validate_resource(resource_json):
    """
    Validates the resource against the FHIR R4 definition and returns resource id.
    """
    if not isinstance(resource_json, dict):
        raise ValueError(f"{RESOURCE_TYPE} resource has to be a JSON object")

    if (resource_json.get('resourceType') or '') != RESOURCE_TYPE:
        raise ValueError(f"resourceType must be {RESOURCE_TYPE}")

    # raises when the payload does not comply with the FHIR R4 definition
    Patient(resource_json)

    return resource_json.get('id') or ''


def validate_request():
    """
    Validates client header and jwt. Returns (clientname, payload).
    """
    clientname = request.headers.get("Client", None) or None
    if not clientname:
        raise ValueError("Invalid Client is passed in header")

    auth_header = request.headers.get("Authorization", None) or None
    if not auth_header or not auth_header.startswith("Bearer "):
        raise ValueError("Unauthorized Token")

    token = auth_header.split(" ")[1]
    if not token:
        logger_obj.log("Missing jwt", 'error')
        raise ValueError("Missing jwt")

    jwt_utils = Jwtutils(clientname=clientname)
    payload = jwt_utils.decode_jwt(token)
    is_valid, diff_list = jwt_utils.validate_jwt_claims(payload)
    if not is_valid:
        logger_obj.log(f"JWT validation failed. Differences in : {diff_list}", 'error')
        raise ValueError(f"JWT validation failed. Differences in : {diff_list}")

    logger_obj.log("valid jwt", 'info')
    return clientname, payload


def get_resource_collection(clientname):
    """
    Attempts to get MongoDB collection. Returns None if DB is unavailable.
    """
    try:
        return get_mongo_db_conn(clientname, COLLECTION_NAME, logger_obj)
    except Exception as e:
        logger_obj.log(f"MongoDB connection unavailable: {str(e)}", 'error')
        return None


@patient_bp.route('/Patient', methods=['POST'])
def create_patient():
    trans_state = 'RequestReceived'

    tracking_data = tracking_data_define()
    tracking_data["MessageType"] = f'{RESOURCE_TYPE}Create'
    tracking_data["Stage"] = trans_state
    tracking_data["RequestReceivedDTS"] = datetime.now()
    tracking_data["RequestURL"] = request.url

    try:
        trans_state = 'VerifyingRequest'
        clientname, payload = validate_request()
        tracking_data["Client"] = clientname

        trans_state = 'ValidatingResource'
        resource_json = get_request_resource()
        resource_id = validate_resource(resource_json)

        resource_collection = get_resource_collection(clientname)

        mongo_document = {
            'InsertDTS': datetime.now(timezone.utc).isoformat(),
            'InsertUser': 'Interopservice',
            'ResourceType': RESOURCE_TYPE,
            'ResourceId': resource_id,
            'ResourceData': resource_json,
            'jwt_payload': payload,
            'DeletedFlag': 0
        }

        if resource_collection is None:
            logger_obj.log("MongoDB unavailable - returning document that would be stored", 'info')
            return jsonify({"mongodb_status": "unavailable", "operation": "create", "document": mongo_document}), 201

        # check for duplicate before creating the resource
        trans_state = 'CheckingDuplicate'
        if resource_id:
            duplicate = resource_collection.find_one({'ResourceId': resource_id, 'DeletedFlag': 0})
            if duplicate:
                tracking_data["Stage"] = trans_state
                tracking_data["Message"] = f"Duplicate {RESOURCE_TYPE} for id {resource_id}"
                tracking_data["SuccessFlag"] = 0
                insert_tracking_record(tracking_data)
                return jsonify({"error": f"{RESOURCE_TYPE} already exists for id {resource_id}"}), 409

        trans_state = 'CreatingResource'
        mongo_document['InsertDTS'] = datetime.now(timezone.utc)
        insert_result = resource_collection.insert_one(mongo_document)

        tracking_data["RequestObjectId"] = str(insert_result.inserted_id)
        tracking_data["Stage"] = 'ResourceCreated'
        tracking_data["Message"] = f"{RESOURCE_TYPE} created"
        tracking_data["SuccessFlag"] = 1
        insert_tracking_record(tracking_data)

        return jsonify(resource_json), 201

    except ValueError as e:
        tracking_data["Stage"] = trans_state
        tracking_data["Message"] = str(e)
        tracking_data["SuccessFlag"] = 0
        insert_tracking_record(tracking_data)
        return jsonify({"error": str(e)}), 400
    except Exception as e:
        logger_obj.log(f"{RESOURCE_TYPE} create Error: {traceback.format_exc()}", 'error')
        tracking_data["Stage"] = trans_state
        tracking_data["Message"] = str(e)
        tracking_data["SuccessFlag"] = 0
        insert_tracking_record(tracking_data)
        return jsonify({"error": f"Error in {RESOURCE_TYPE} create: {str(e)}"}), 500


@patient_bp.route('/Patient/<resource_id>', methods=['PUT'])
def update_patient(resource_id):
    trans_state = 'RequestReceived'

    tracking_data = tracking_data_define()
    tracking_data["MessageType"] = f'{RESOURCE_TYPE}Update'
    tracking_data["Stage"] = trans_state
    tracking_data["RequestReceivedDTS"] = datetime.now()
    tracking_data["RequestURL"] = request.url

    try:
        trans_state = 'VerifyingRequest'
        clientname, payload = validate_request()
        tracking_data["Client"] = clientname

        trans_state = 'ValidatingResource'
        resource_json = get_request_resource()
        payload_resource_id = validate_resource(resource_json)

        if payload_resource_id and payload_resource_id != resource_id:
            raise ValueError("Resource id in the body does not match the id in the URL")

        resource_json['id'] = resource_id

        resource_collection = get_resource_collection(clientname)

        update_document = {
            'ResourceData': resource_json,
            'UpdateDTS': datetime.now(timezone.utc).isoformat(),
            'UpdateUser': 'Interopservice',
            'jwt_payload': payload
        }

        if resource_collection is None:
            logger_obj.log("MongoDB unavailable - returning document that would be stored", 'info')
            return jsonify({"mongodb_status": "unavailable", "operation": "update", "resource_id": resource_id, "document": update_document}), 200

        trans_state = 'UpdatingResource'
        update_document['UpdateDTS'] = datetime.now(timezone.utc)
        update_result = resource_collection.update_one(
            {'ResourceId': resource_id, 'DeletedFlag': 0},
            {'$set': update_document}
        )

        if update_result.matched_count == 0:
            tracking_data["Stage"] = trans_state
            tracking_data["Message"] = f"{RESOURCE_TYPE} not found for id {resource_id}"
            tracking_data["SuccessFlag"] = 0
            insert_tracking_record(tracking_data)
            return jsonify({"error": f"{RESOURCE_TYPE} not found for id {resource_id}"}), 404

        tracking_data["Stage"] = 'ResourceUpdated'
        tracking_data["Message"] = f"{RESOURCE_TYPE} updated"
        tracking_data["SuccessFlag"] = 1
        insert_tracking_record(tracking_data)

        return jsonify(resource_json), 200

    except ValueError as e:
        tracking_data["Stage"] = trans_state
        tracking_data["Message"] = str(e)
        tracking_data["SuccessFlag"] = 0
        insert_tracking_record(tracking_data)
        return jsonify({"error": str(e)}), 400
    except Exception as e:
        logger_obj.log(f"{RESOURCE_TYPE} update Error: {traceback.format_exc()}", 'error')
        tracking_data["Stage"] = trans_state
        tracking_data["Message"] = str(e)
        tracking_data["SuccessFlag"] = 0
        insert_tracking_record(tracking_data)
        return jsonify({"error": f"Error in {RESOURCE_TYPE} update: {str(e)}"}), 500


@patient_bp.route('/Patient/<resource_id>', methods=['DELETE'])
def delete_patient(resource_id):
    trans_state = 'RequestReceived'

    tracking_data = tracking_data_define()
    tracking_data["MessageType"] = f'{RESOURCE_TYPE}Delete'
    tracking_data["Stage"] = trans_state
    tracking_data["RequestReceivedDTS"] = datetime.now()
    tracking_data["RequestURL"] = request.url

    try:
        trans_state = 'VerifyingRequest'
        clientname, payload = validate_request()
        tracking_data["Client"] = clientname

        resource_collection = get_resource_collection(clientname)

        delete_document = {
            'DeletedFlag': 1,
            'DeletedDTS': datetime.now(timezone.utc).isoformat(),
            'DeletedUser': 'Interopservice'
        }

        if resource_collection is None:
            logger_obj.log("MongoDB unavailable - returning document that would be stored", 'info')
            return jsonify({"mongodb_status": "unavailable", "operation": "delete", "resource_id": resource_id, "document": delete_document}), 200

        # delete is a soft delete, the existing resource is marked as deleted
        trans_state = 'DeletingResource'
        delete_document['DeletedDTS'] = datetime.now(timezone.utc)
        delete_result = resource_collection.update_one(
            {'ResourceId': resource_id, 'DeletedFlag': 0},
            {'$set': delete_document}
        )

        if delete_result.matched_count == 0:
            tracking_data["Stage"] = trans_state
            tracking_data["Message"] = f"{RESOURCE_TYPE} not found for id {resource_id}"
            tracking_data["SuccessFlag"] = 0
            insert_tracking_record(tracking_data)
            return jsonify({"error": f"{RESOURCE_TYPE} not found for id {resource_id}"}), 404

        tracking_data["Stage"] = 'ResourceDeleted'
        tracking_data["Message"] = f"{RESOURCE_TYPE} deleted"
        tracking_data["SuccessFlag"] = 1
        insert_tracking_record(tracking_data)

        return '', 204

    except ValueError as e:
        tracking_data["Stage"] = trans_state
        tracking_data["Message"] = str(e)
        tracking_data["SuccessFlag"] = 0
        insert_tracking_record(tracking_data)
        return jsonify({"error": str(e)}), 400
    except Exception as e:
        logger_obj.log(f"{RESOURCE_TYPE} delete Error: {traceback.format_exc()}", 'error')
        tracking_data["Stage"] = trans_state
        tracking_data["Message"] = str(e)
        tracking_data["SuccessFlag"] = 0
        insert_tracking_record(tracking_data)
        return jsonify({"error": f"Error in {RESOURCE_TYPE} delete: {str(e)}"}), 500


@patient_bp.route('/Patient', methods=['GET'])
def search_patient():
    trans_state = 'RequestReceived'

    tracking_data = tracking_data_define()
    tracking_data["MessageType"] = f'{RESOURCE_TYPE}Search'
    tracking_data["Stage"] = trans_state
    tracking_data["RequestReceivedDTS"] = datetime.now()
    tracking_data["RequestURL"] = request.url

    try:
        trans_state = 'VerifyingRequest'
        clientname, payload = validate_request()
        tracking_data["Client"] = clientname

        trans_state = 'SearchingResource'
        search_query = {'DeletedFlag': 0}

        resource_id = request.args.get('_id', None) or None
        if resource_id:
            search_query['ResourceId'] = resource_id

        identifier = request.args.get('identifier', None) or None
        if identifier:
            search_query['ResourceData.identifier.value'] = identifier

        resource_collection = get_resource_collection(clientname)

        if resource_collection is None:
            logger_obj.log("MongoDB unavailable - cannot perform search", 'info')
            return jsonify({"mongodb_status": "unavailable", "operation": "search", "query": search_query, "message": "MongoDB is not connected. Search cannot return results."}), 200

        matching_records = resource_collection.find(
            filter=search_query,
            projection={'ResourceData': 1, '_id': 0}
        )

        entries = [
            {'resource': record.get('ResourceData')}
            for record in matching_records if record.get('ResourceData')
        ]

        tracking_data["Stage"] = 'ResourceSearched'
        tracking_data["Message"] = f"{RESOURCE_TYPE} search returned {len(entries)} records"
        tracking_data["SuccessFlag"] = 1
        insert_tracking_record(tracking_data)

        # search response is returned as a FHIR searchset Bundle
        return jsonify({
            'resourceType': 'Bundle',
            'type': 'searchset',
            'total': len(entries),
            'entry': entries
        }), 200

    except ValueError as e:
        tracking_data["Stage"] = trans_state
        tracking_data["Message"] = str(e)
        tracking_data["SuccessFlag"] = 0
        insert_tracking_record(tracking_data)
        return jsonify({"error": str(e)}), 400
    except Exception as e:
        logger_obj.log(f"{RESOURCE_TYPE} search Error: {traceback.format_exc()}", 'error')
        tracking_data["Stage"] = trans_state
        tracking_data["Message"] = str(e)
        tracking_data["SuccessFlag"] = 0
        insert_tracking_record(tracking_data)
        return jsonify({"error": f"Error in {RESOURCE_TYPE} search: {str(e)}"}), 500
