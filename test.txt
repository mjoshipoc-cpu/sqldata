@patinetcreate_bp.route('/fhit/patient', methods=['POST'])
def create_patient():
    try:
        trans_state = 'RequestReceived'

        tracking_data = tracking_data_define()
        tracking_data["Stage"] = trans_state
        tracking_data["RequestReceivedDTS"] = datetime.now()

        tracking_data["RequestURL"] = request.url
        # verify the requst
        trans_state = 'VerifyingRequest'

        # get client from header
        clientname = request.headers.get("Client", None) or None
        if not clientname:
            return {"error": "Invalid Client is passed in header"}, 400
        tracking_data["Client"] = clientname
        jwt_utils = Jwtutils(clientname=clientname)

        # get and validate the header authorization
        auth_header = request.headers.get("Authorization", None) or None

        if not auth_header or not auth_header.startswith("Bearer "):
            return {"error": "Unauthorized Token"}, 401

        token = auth_header.split(" ")[1]

        if not token:
            logger_obj.log("Missing jwt", 'error')
            return jsonify({"error": "Missing jwt"}), 401

        # verify the request
        if not request.is_json:
            logger_obj.log("Request is not JSON", 'error')
            raise ValueError("Request has to be JSON")

        # check the content type
        content_type = request.content_type.split('/')[1]
        logger_obj.log(content_type, 'debug')
        if content_type not in ['json']:
            logger_obj.log("Unsupported content type", 'info')
            raise ValueError("Unsupported content type or invalid data format")

        # get and validate the data
        json_data = json.loads(request.get_data())

        if json_data is None:
            raise ValueError("no data submitted in the body")

        # validating the jwt
        trans_state = 'ValidatingJWT'
        payload = jwt_utils.decode_jwt(token)
        # clientname = payload['sub']

        is_valid, diff_list = jwt_utils.validate_jwt_claims(payload)
        if not is_valid:
            logger_obj.log(f"JWT validation failed. Differences in : {diff_list}", 'error')
            raise ValueError(f"JWT validation failed. Differences in : {diff_list}")

        logger_obj.log("valid jwt", 'info')

        trans_state = 'InitializingBundleObject'
        # Initialize the Bundle object
        patient = Patient(json_data)

        # creating mongo db collection objects for request and response
        request_collection = get_mongo_db_conn(clientname, 'Patient', logger_obj)
        request_result = request_collection.insert_one({
            'InsertDTS': datetime.now(),
            'InsertUser': 'Interopservice',
            'RequestResourceType': 'Patient',
            'RequestDataFromSource': patient,  # data from EMR

        })
        request_obj_id = ObjectId(request_result.inserted_id)
        return jsonify({"id": request_obj_id, "message": "Patient created successfully"}), 201
    except Exception as e:
        return jsonify({"error": str(e), "traceback": traceback.format_exc()}), 400
