def create_app():

    app.register_blueprint(inquiry_bp,url_prefix=prefix)
    app.register_blueprint(subscription_bp,url_prefix=prefix)
    app.register_blueprint(response_bp,url_prefix=prefix)
    app.register_blueprint(downloadattachment_bp,url_prefix=prefix)
    app.register_blueprint(testing_bp,url_prefix=prefix)
    app.register_blueprint(crupdate_bp,url_prefix=prefix)
    app.register_blueprint(crattachmentqueue_bp,url_prefix=prefix)
    app.register_blueprint(patient_bp,url_prefix=prefix)
    app.register_blueprint(practitioner_bp,url_prefix=prefix)
    app.register_blueprint(organization_bp,url_prefix=prefix)
    app.register_blueprint(location_bp,url_prefix=prefix)
    app.register_blueprint(careteam_bp,url_prefix=prefix)

    register_error_handlers(app)
    return app

app = create_app()

from flask import jsonify

def register_error_handlers(app):
    @app.errorhandler(ValueError)
    def handle_value_error(e):
        return jsonify({'error': str(e)}), 400

    @app.errorhandler(Exception)
    def handle_exception(e):
        return jsonify({'error': 'Internal server error', 'details': str(e)}), 500
