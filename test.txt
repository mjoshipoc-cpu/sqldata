from mapping.PASModule.claimrequest.memberdetails import MemberDetails
from mapping.PASModule.claimrequest.requestingproviderdetails import RequestingProviderDetails
from mapping.PASModule.claimrequest.servicedetails import ServiceDetails
from utils.logutils import logger
import traceback


class claimUpdate:
    def __init__(self, bundle,ruleparser, update_reference_details, memberId, passeqmapping_record):
        self.bundle = bundle
        self.ruleparser = ruleparser
        self.update_reference_details = update_reference_details
        self.memberId = memberId
        self.passeqmapping_record = passeqmapping_record
        self.logger_obj = logger()
    
    def main(self):

        try:
            claim_resource = self.ruleparser.get_resource('claim')
            certificationType = self.ruleparser.get_extension_value(claim_resource.extension, 'extension-certificationType')
            self.logger_obj.log(f"certificationType: {certificationType}")

            certificationTypecode = ""
            certificatedescription = ""

            if certificationType and isinstance(certificationType, list) and len(certificationType) > 0:
                certificationTypecode = certificationType[0].get("code", "") or ""
                certificatedescription = certificationType[0].get("display", "") if isinstance(certificationType[0], dict) else ""

            if (certificationTypecode == "" and certificatedescription == "" ) or (certificationTypecode != '3' or certificatedescription.lower() != 'cancel'):
                print(f"certificationType code or description is not cancel: {certificationTypecode}, {certificatedescription}")
                #return None
            certificateTypeDict = {
                "code": certificationTypecode,
                "description": certificatedescription
            }
            self.logger_obj.log('before member details')
            memberdetails = MemberDetails(self.ruleparser,self.memberId).memberdetailsmapping()
            self.logger_obj.log('after member details')
            requestingProviderDetails = RequestingProviderDetails(self.ruleparser).requestingprovidermapping()
            self.logger_obj.log('after requesting provider details')
            referenceDetails = self.update_reference_details
            self.logger_obj.log('after reference details')
            authorizationdetails = self.ruleparser.get_authorization_details(certificateTypeDict)
            self.logger_obj.log('after authorization details')
            contactDetails = self.ruleparser.get_contact_details()
            self.logger_obj.log('after contact details')
            diagnosisdetails = self.ruleparser.get_diagnosis_details()
            self.logger_obj.log('after diagnosis details')
            relatedConditiondetails = self.ruleparser.get_related_condition_details()
            self.logger_obj.log('after related condition details')
            servicedetails = ServiceDetails(self.ruleparser,self.passeqmapping_record).get_services_updated()
            self.logger_obj.log('after service details')
            supportinginformation = self.ruleparser.get_authlevel_supportinginfo()
            self.logger_obj.log('after supporting info details')

            outputdata = {"authorization":{
                "referenceDetails": referenceDetails,
                "authorizationDetails": authorizationdetails,
                "contactDetails" : contactDetails,
                "memberDetails": memberdetails,
                "diagnosisDetails": diagnosisdetails,
                "requestingProviderDetails": requestingProviderDetails,
                "relatedConditions": relatedConditiondetails,
                "services": servicedetails,
                "supportingInformation": supportinginformation
            }}  
            return outputdata
        except Exception as e:
            self.logger_obj.log(f"Claimupdate main Error: {traceback.format_exc()} ")
            raise ValueError(f"Error building Claimupdate: {str(e)}")
