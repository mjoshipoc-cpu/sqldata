class ClaimResponseBuilder:
    def __init__(self, input_json,claim_request_bundle, claim_rules):
        self.input_json = input_json
        self.auth = input_json.get('authorization', {}) or {}
        self.claim_request_bundle = claim_request_bundle
        self.claim_rules = claim_rules
        #self.claim_resource = self.claim_rules.get_resource('Claim')
        referred_resource = self.claim_rules.get_resources_referenced_in_claim()
        self.bundle_resources = []
        self.patient_resource = referred_resource.get('patient')
        self.insurer_resource = referred_resource.get('insurer')
        self.claim_resource = referred_resource.get('claim')
        self.insurer_ref = self.claim_resource.insurer.reference
        self.patient_ref = self.claim_resource.patient.reference
        self.logger_obj = logger()
        
    def _safe_get(self, d, keys):
        current = d
        for key in keys:
            if current is None:
                return None
            if isinstance(current, dict):
                current = current.get(key)
            else:
                return None
        return current

    
    def build_claimresponse(self):
        cr = claimresponse.ClaimResponse()
        cr.use = self.claim_resource.use
        cr.status = 'active'
        cr.request = fhirreference.FHIRReference({'reference': self.claim_resource.id})
        cr.created = fhirdatetime.FHIRDateTime(datetime.now(timezone.utc).strftime('%Y-%m-%dT%H:%M:%SZ'))
        
        
        cr.patient = fhirreference.FHIRReference({'reference': self.patient_ref})
        cr.insurer = fhirreference.FHIRReference({'reference': self.insurer_ref})
        if not self.auth:
            cr.outcome = 'error'
            cr.type = self.claim_resource.type
        else:
            cr.outcome = self._safe_get(self.auth, ['authorizationDetails', 'authOutcome']) or ''
            # Identifier
            ref = self.auth.get('referenceDetails', {}) or {}
            system = ref.get('pasIdentifierSystem','') or ''
            value = ref.get('pasIdentifierValue','') or ''
            if system and value:
                cr.identifier = [identifier.Identifier({
                    "system": "http://exl.com/careradius",
                    "value": f"{value}-response"
                })]
            cr.preAuthRef = ref.get('referenceNumber', '') or ''    

            # Type
            authDetails = self.auth.get('authorizationDetails') or {}
            caseType = authDetails.get('caseType') or {}
            caseTypeDisplay = caseType.get('description') or ''
            cr.type = codeableconcept.CodeableConcept({
                "coding": [{"code": "", "display": caseTypeDisplay}]
            })
            

            # PreAuthPeriod 
            start = self._safe_get(self.auth, ['authorizationDetails', 'dateValidFrom'])
            end = self._safe_get(self.auth, ['authorizationDetails', 'dateValidTo'])
            if start or end:
                pre_auth_period = period.Period()
                pre_auth_period.start = fhirdatetime.FHIRDateTime(start)
                pre_auth_period.end = fhirdatetime.FHIRDateTime(end)
                cr.preAuthPeriod = pre_auth_period
            # Adjudication
            status = self._safe_get(self.auth, ['authorizationDetails', 'status']) or ''
            itemadj = claimresponse.ClaimResponseItemAdjudication()
            itemadj.category = codeableconcept.CodeableConcept({
                "coding": [{"code": "submitted"}]
            })
            # Add extension slice for reviewAction
            review_action_ext = extension.Extension()
            review_action_ext.url = "http://example.org/fhir/StructureDefinition/reviewAction"
            review_action_ext.valueCodeableConcept = codeableconcept.CodeableConcept({
                "coding": [{"code": status}]
            })
            itemadj.extension = [review_action_ext]
            cr.adjudication = [itemadj]


            
            #process notes
            auth_processnotes = self.auth.get('processNotes')
            if auth_processnotes:
                process_notes, crseq_uniquenum_map = self.process_notes(cr, auth_processnotes)
                cr.processNote = process_notes


            # Items
            cr.item = []
            cr.addItem = []
            services = self.auth.get('services', [])
            for i, svc in enumerate(services, 1):
                item = claimresponse.ClaimResponseItem()
                item.extension = []

                additem = claimresponse.ClaimResponseAddItem()
                additem.extension = []

                pas_seq=""
                generated_pas_seq = ""

                self.logger_obj.log(f"Processing service {i} with data: {svc}")
                if svc["servicePASMapping"]["pasSequence"]:
                    pas_seq = svc["servicePASMapping"]["pasSequence"]
                    generated_pas_seq = svc["servicePASMapping"]["generatedPasSequence"] if pas_seq == "-1" else ""

                cr_seq = self._safe_get(svc, ['servicePASMapping', 'crSequence'])
                
                if pas_seq != "-1":
                    self.logger_obj.log(f"Setting itemSequence for item with pas_seq {pas_seq}")
                    item.itemSequence = int(pas_seq)

                else:
                    self.logger_obj.log(f"Setting itemSequence for additem with pas_seq {generated_pas_seq}")
                    additem.itemSequence = [int(generated_pas_seq)]

                # Adjudication
                status = self._safe_get(svc, ['statusDetails', 'status']) 
                if status in ['A3', 'A4', 'CT']:
                    if pas_seq != "-1":
                        administrationReferenceNumber_ext = self.claim_rules.create_extension('administrationReferenceNumber', 'String', {'value':cr_seq})
                        
                        item.extension.append(administrationReferenceNumber_ext)
                    else:
                        administrationReferenceNumber_ext = self.claim_rules.create_extension('administrationReferenceNumber', 'String', {'value':cr_seq})
                        additem.extension.append(administrationReferenceNumber_ext)

                if status in ['A1', 'A2', 'A6']:
                    if pas_seq != "-1":
                        previousAuthorizationNumber_ext = self.claim_rules.create_extension('previousAuthorizationNumber', 'String', {'value':cr_seq})
                        item.extension.append(previousAuthorizationNumber_ext)
                    else:
                        previousAuthorizationNumber_ext = self.claim_rules.create_extension('previousAuthorizationNumber', 'String', {'value':cr_seq})
                        additem.extension.append(previousAuthorizationNumber_ext)
                
                

                
                # service procedures
                self.process_service_procedures(svc, item, pas_seq,additem)

                # authoried providers
                self.build_authorized_providers(svc, item, pas_seq, additem)

                #authorized quantity details
                self.add_quantity_details(svc, item, pas_seq, additem)

                # authorized modifiers
                self.build_authorized_modifiers(svc, item, pas_seq, additem)
                #authorized service dates
                self.build_service_dates(svc, item, pas_seq, additem)

                # status details
                self.build_status_details(svc, item, pas_seq, additem)

                


                if item is not None and pas_seq != "-1":
                    if auth_processnotes:
                        crseq_uniquenum_map1 = [m for m in crseq_uniquenum_map if m.get("crSequence") == cr_seq]
                        if crseq_uniquenum_map1:
                            item.noteNumber = [m["uniqueNumber"] for m in crseq_uniquenum_map1]
                    cr.item.append(item)
                if additem is not None and pas_seq == "-1":
                    self.logger_obj.log(f"Adding additem with itemSequence {additem.itemSequence}")
                    cr.addItem.append(additem)

        #errors
        self.build_auth_errors(self.input_json,cr)

        

        return cr

    def add_quantity_details(self, svc, item, pas_seq, additem=None):
        qty_details = svc.get('authorizedQuantityDetails', {})
        try:
            quantity = float(qty_details.get('quantity', 0.0)) if qty_details.get('quantity') is not None else 0.0
        except (ValueError, TypeError):
            quantity = 0.0

        quantity_qualifier = qty_details.get('quantityQualifier', "") or ""

        # If it's a "-1" item, map directly to addItem.quantity
        if pas_seq == "-1" and additem is not None:
            additem.quantity = quantity_.Quantity({
                "value": quantity,
                "code": quantity_qualifier
            })
            return

        # Otherwise, map inside authorizedItemDetail extension
        if item is not None:
            # Build valueQuantity dictionary
            value_quantity = {
                "value": quantity,
                "code": quantity_qualifier  }

            # Create nested extension using create_extension for valuePASQuantity
            value_pas_quantity_ext = self.claim_rules.create_extension(
                url="quantity",
                value_type="Quantity",
                value=value_quantity,
                fullurl=False
            )
            self.claim_rules.add_ext_authorizeditemdetails(item,value_pas_quantity_ext)
        return

    def build_authorized_modifiers(self, svc, item, pas_seq, additem):
        """
        Adds authorized modifiers to either ClaimResponse.item.extension or addItem.modifier,
        depending on pas_seq.
        
        Args:
            svc (dict): Service object from input JSON.
            item: FHIR ClaimResponse.item instance.
            additem: FHIR ClaimResponse.addItem instance.
            pas_seq (str): The pasSequence value to determine where to add the modifier.
        """
        modifier_details = svc.get("authorizedModifierDetails", [])
        if not modifier_details:
            return

        modifier_codings = []
        for mod in modifier_details:
            code = mod.get("code", "") or ""
            display = mod.get("description", "") or ""
            if code is not None or display is not None:
                modifier_codings.append({
                    "code": code,
                    "display": display
                })
        if not modifier_codings:
            return

        # Construct CodeableConcept object
        modifier_cc = codeableconcept.CodeableConcept({
            "coding": modifier_codings
        })

        if pas_seq == "-1" and additem:
            # Add to addItem.modifier
            additem.modifier = [modifier_cc]

        elif item:
            # Create modifier extension using helper
            modifier_ext = self.claim_rules.create_extension(
                url="modifier",
                value_type="CodeableConcept",
                value=modifier_codings,
                fullurl=False,
                displayrequired=True
            )

            self.claim_rules.add_ext_authorizeditemdetails(item,modifier_ext)
        return

    def build_service_dates(self, svc, item, pas_seq, additem):
        """
        Add servicedDate or servicedPeriod depending on date values.
        Uses `create_extension` for item extensions.
        """
        service_details = svc.get("authorizedServiceDetails", {"serviceDateFrom": "2025-06-15", "serviceDateTo": "2025-06-15"}) or None
        if service_details is not None:
            date_from = service_details.get("serviceDateFrom","") or ""
            date_to = service_details.get("serviceDateTo","") or ""

            #if not date_from and not date_to:
            #    return

            # For new PAS sequence, use servicedDate/servicedPeriod
            if pas_seq == "-1" and additem and (date_from or date_to):
                if date_from == date_to:
                    additem.servicedDate = fhirdate.FHIRDate(date_from)
                else:
                    period1 = period.Period()
                    if date_from:
                        period1.start = fhirdatetime.FHIRDateTime(date_from)
                    if date_to:
                        period1.end = fhirdatetime.FHIRDateTime(date_to)
                    additem.servicedPeriod = period1

            elif item and (date_from or date_to):
                # Use extension
                if date_from == date_to:
                    ext = self.claim_rules.create_extension("itemRequestedServiceDate", "date", date_from)
                else:
                    ext = self.claim_rules.create_extension("itemRequestedServiceDate", "period", {
                        "start": date_from,
                        "end": date_to
                    })

                if not item.extension:
                    item.extension = []
                item.extension.append(ext)
            return

    def build_status_details(self, svc, item, pas_seq, additem):
        """
        Populates adjudication.reviewAction extension and preAuthIssueDate on item/additem based on statusDetails.
        """
        status_details = svc.get("statusDetails", {})
        status = status_details.get("status","") or ""
        reason = status_details.get("determinationReason",None) or None
        date_changed = status_details.get("dateStatusChanged",None) or None
        time_changed = status_details.get("timeStatusChanged",None) or None
        if time_changed is not None:
            # Parse the HHMM string into a datetime object
            time_obj = datetime.strptime(time_changed, '%H%M')

            # Convert it to HH:MM:SS format
            formatted_time = time_obj.strftime('%H:%M:%S')

        cr_target = additem if pas_seq == "-1" else item

        if cr_target is None:
            return

        if date_changed is not None or time_changed is not None:
            
            combined = f"{date_changed}T{formatted_time}.000Z" if date_changed is not None and time_changed is not None else date_changed or time_changed
            #dt = datetime.strptime(combined, "%Y-%m-%d %H:%M:%S")
            preauth_ext = extension.Extension()
            preauth_ext.url = "http://hl7.org/fhir/us/davinci-pas/StructureDefinition/extension-itemPreAuthIssueDate"
            preauth_ext.valueDateTime = fhirdatetime.FHIRDateTime(combined)
            if not cr_target.extension:
                cr_target.extension = []
            cr_target.extension.append(preauth_ext)

        # --- adjudication.extension.reviewAction ---
        if status is not None or reason is not None:
            adjudication = claimresponse.ClaimResponseItemAdjudication()
            review_action_ext = extension.Extension()
            review_action_ext.url = "http://hl7.org/fhir/us/davinci-pas/StructureDefinition/extension-reviewAction"
            review_action_ext.extension = []

            adjudication.category = codeableconcept.CodeableConcept({
                "coding": [{
                    "system": "http://terminology.hl7.org/CodeSystem/adjudication",
                    "code": "submitted"
                }]
            })

            # code
            if status is not None:
                review_action_ext.extension.append(
                    self.claim_rules.create_extension("reviewAction", "CodeableConcept", {"code": status},displayrequired=False)
                )
                

            # reasonCode
            if reason is not None:
                review_action_ext.extension.append(
                    self.claim_rules.create_extension("reasonCode", "CodeableConcept", {"code": reason}, displayrequired=False,fullurl=False)
                )

            adjudication.extension = [review_action_ext]
            cr_target.adjudication = cr_target.adjudication or []
            cr_target.adjudication.append(adjudication)
        return



    def process_service_procedures(self, svc, item, pas_seq, add_item=None):
        """
        Processes `serviceProcedures` from a service entry:
        - Adds authorizedItemDetail extension to item.extension (if pas_seq != "-1")
        - Adds procedure code to add_item (if pas_seq == "-1")
        """
        procedures = svc.get('serviceProcedures', [])
        if not procedures:
            return

        low_codings = []
        high_codings = []
        

        for proc in procedures:
            details = proc.get('authorisedProcedureDetails', {})
            low = details.get('procedureLow', {})
            high = details.get('procedureHigh', {})

            low_code = low.get("code")
            low_display = low.get("shortDescription")
            high_code = high.get("code")
            high_display = high.get("shortDescription")

            low_codings.append({
                "code": low_code or "",
                "display": low_display or ""
            })
            high_codings.append({
                "code": high_code or "",
                "display": high_display or ""
            })
        if str(pas_seq).lower() == "-1" and add_item is not None:
            # PAS Sequence = "-1" ? populate addItem
            if low_codings is not None:
                add_item.productOrService = codeableconcept.CodeableConcept()
                add_item.productOrService.coding = [coding.Coding(c) for c in low_codings]

            if high_codings is not None:
                ext = self.claim_rules.create_extension(
                    url="productOrServiceCodeEnd",
                    value_type="CodeableConcept",
                    value=high_codings,
                    displayrequired=True,
                    fullurl=False
                )
                
                add_item.extension.append(ext)

        else:
            # PAS Sequence =! "-1"  populate item.extension with authorizedItemDetail
            authorized_detail_ext = extension.Extension()
            authorized_detail_ext.url = "http://hl7.org/fhir/us/davinci-pas/StructureDefinition/extension-authorizedItemDetail"
            authorized_detail_ext.extension = []
            
            if low_codings is not None:
                ext_low = self.claim_rules.create_extension(
                    url="productOrServiceCode",
                    value_type="CodeableConcept",
                    value=low_codings,
                    displayrequired=True,
                    fullurl=False
                )
                authorized_detail_ext.extension.append(ext_low)

            if high_codings is not None:
                ext_high = self.claim_rules.create_extension(
                    url="productOrServiceCodeEnd",
                    value_type="CodeableConcept",
                    value=high_codings,
                    displayrequired=True,
                    fullurl=False
                )

                authorized_detail_ext.extension.append(ext_high)
            # Only add to item.extension if at least one inner extension exists
            if authorized_detail_ext.extension:
                item.extension = item.extension or []
                item.extension.append(authorized_detail_ext)

    def build_auth_errors(self,input_data,cr):
        
        errors = []
        cr_errors = input_data.get("errors", []) or None
        if cr_errors is None or len(cr_errors) == 0:
            return
        for err in input_data.get("errors", []):
            error = claimresponse.ClaimResponseError()
            cr_seq = err.get("crSequence", "") or ""
            if cr_seq:
                pas_seq = self.claim_rules.get_pas_seq(input_data, cr_seq) or None
                if pas_seq:
                    error.itemSequence = pas_seq
                    
            # Reject reason (CodeableConcept)
            rr_cc = self.claim_rules.dict_to_codeableconcept(err.get("rejectReason", {}))
            if rr_cc:
                error.code = rr_cc
            

            # followUpAction extension (valueCodeableConcept)
            fua_codings = []
            fua_code = err.get("followUpAction", {}).get("code")
            fua_display = err.get("followUpAction", {}).get("description")
            fua_codings.append({
            "code": fua_code or "",
            "display": fua_display or ""
                })
            fua = self.claim_rules.create_extension(
                url="errorFollowUpAction",
                value_type="CodeableConcept",
                value=fua_codings,
                displayrequired=True,
                fullurl=False
            )

            error.extension = [fua]

            errors.append(error)

        cr.error = errors

        return
    

    def process_notes(self, cr, auth_processnotes):

        process_notes = []
        crseq_uniquenum_map = []
        for i in auth_processnotes:
            crseq = i.get("crSequence", "") or ""
            text = i.get("text", "") or ""

            unique_number = self.claim_rules.get_unique_positive_int32()
            process_note = claimresponse.ClaimResponseProcessNote()

            process_note.number = unique_number
            process_note.text = text
            process_notes.append(process_note)
            crseq_uniquenum_map.append({
                "crSequence": crseq,
                "uniqueNumber": unique_number
            })
        
        return process_notes, crseq_uniquenum_map





    def build_bundle(self):
        
        # Parse ClaimResponse JSON into a FHIR resource object
        claim_resp = self.build_claimresponse()
        self.bundle_resources.append(claim_resp)
        
        self.claim_rules.add_resource_to_bundle(self.bundle_resources,self.patient_resource)
        self.claim_rules.add_resource_to_bundle(self.bundle_resources,self.insurer_resource)
        self.claim_rules.add_resource_to_bundle(self.bundle_resources,self.claim_resource)

        # Create a Bundle of type "searchset" or "collection"
        b = bundle.Bundle()
        b.id = str(uuid.uuid4())
        b.type = "collection"  # use "document", "searchset", etc., as needed
        now_iso = datetime.now(timezone.utc).isoformat()
        b.timestamp = fhirinstant.FHIRInstant(now_iso)
        b.entry=[]

        def derive_full_url(resource):
            if hasattr(resource, 'resource_type'):
                resource_type = resource.resource_type
                
                resource_id = getattr(resource, 'id', None)

                if resource_id:
                    return f"{resource_type}/{resource_id}"
                else:
                    # Generate UUID and assign it as the resource ID
                    generated_id = str(uuid.uuid4())
                    resource.id = generated_id
                    return f"urn:uuid:{generated_id}"
            else:
                raise ValueError("Invalid FHIR resource passed to derive_full_url")

        seen_urls = set()
        for resource in getattr(self, 'bundle_resources', []):
            full_url = derive_full_url(resource)
            if full_url not in seen_urls:
                entry = bundle.BundleEntry()
                entry.fullUrl = full_url
                entry.resource = resource
                b.entry.append(entry)
                seen_urls.add(full_url)
        # Set total count
        b.total = len(b.entry)

        return b
