        #content_type = request.content_type
        if content_type and 'application/json' in content_type:
            try:
                data = request.get_json(force=True)
                return 'json', data
            except Exception:
                return 'invalid', None
        elif content_type and ('application/xml' in content_type or 'text/xml' in content_type):
            try:
                data = ET.fromstring(request.data)
                return 'xml', data
            except Exception:
                return 'invalid', None
        else:
            return 'unknown', None

def query_configdb(query, params=None):
    """
    Executes a query on the configuration database.
    
    :param query: SQL query to execute.
    :param params: Parameters for the SQL query (tuple or list).
    :return: Query result as a list of dictionaries.

    Query with one condition
    query = "SELECT * FROM config_table WHERE key = %s"
    params = ('api_timeout',)
    result = query_configdb(query, params)

    Multiple parameters
    query = "SELECT * FROM config_table WHERE key = %s AND environment = %s"
    params = ('api_timeout', 'prod')
    result = query_configdb(query, params)

    LIKE query
    query = "SELECT * FROM config_table WHERE key LIKE %s"
    params = ('%timeout%',)
    result = query_configdb(query, params)

    """
    conn = None
    try:
        conn = get_postgres_conn()
        with conn.cursor(cursor_factory=RealDictCursor) as cur:
            cur.execute(query, params)
            rows = cur.fetchall()

            if len(rows) == 1:
                first_row = rows[0]
                if len(first_row) == 1:
                    # One row, one column ? return just the value
                    return list(first_row.values())[0]
                else:
                    # One row, multiple columns ? return single dict
                    return first_row
            else:
                # Multiple rows ? return list of dicts
                return rows
    except Exception as e:
        raise e
    finally:
        if conn:
            conn.close()

    TEST_CHARACTER_SET = 'iso-8859-1'

    # print ("Base dir is " + basedir)

    def __init__(self, keyString=None):
       self.pathToKeyfile = "{basedir}/interop.key".format(basedir=basedir)
       if not keyString is None:
            self._getFernet(keyString)
       else:
            self._getFernet()
            if not self.fernetInstance == None and not self._testRoundTrip():
                print('ERROR: something is wrong with InteropEncrypt. test failed.')

    def _getFernet(self, keyString=None):
        '''
        Read the key file and remember the Fernet instance.
        input: self.pathToKeyfile
        output: self.fernetInstance - the fernet object on which encrypt/decrypt is called
        '''
        if keyString is not None:
            self.keyBytes = keyString
            self.fernetInstance = Fernet(self.keyBytes)
        else:
            try:
                with open(self.pathToKeyfile, "rb") as keyfile:
                    self.keyBytes = keyfile.read()
                self.fernetInstance = Fernet(self.keyBytes)
            #except FileNotFoundError as fnf:
            #    print('ERROR: Keyfile=[{}] not found!'.format(self.pathToKeyfile))
            #    self.keyBytes = None
            #    self.fernetInstance = None
            except Exception as e:
                print(e)
                print('ERROR: {}'.format(e.strerror))
                self.keyBytes = None
                self.fernetInstance = None
        return

    def _fileExists(self, path):
        try:
            fh = open(path,"r")
        #except FileNotFoundError as fnf:
        except Exception as fnf:
            return False
        return True

    def _testRoundTrip(self,testString=None):
        '''Does an encrypt >> decrypt roundtrip and tests to make sure no data is lost.
        Test with all 256 extended ascii (iso-8859-1) characters.
        NOTE: encrypt() also does a compare so this is a bit redundant.'''
        if testString == None:
            testString='''this is a password string
            !@#$%^&*()_+ 01234567890,.{}[]|\;':" '''
            #testString=codecs.decode(bytes([n for n in range(256)]),InteropEncrypt.TEST_CHARACTER_SET)
            testString=codecs.decode(bytearray([n for n in range(127)]), InteropEncrypt.TEST_CHARACTER_SET)
            testString=str(testString)      #convert from unicode to str
        x = self.decrypt(self.encrypt(testString))
        return True if x == testString else False
            
    def createKeyFile(self, overwrite=False):
        keyBytes_new = Fernet.generate_key()
        if keyBytes_new == self.keyBytes:
            print('WARNING: new key is same as the existing key.  no changes made.')
            return
        elif self._fileExists(self.pathToKeyfile) and not overwrite:
            print('WARNING: keyfile exists, must specify -overwrite to replace')
            return
        else:
            with open(self.pathToKeyfile,"wb") as keyfile:
                keyfile.write(keyBytes_new)
        self._getFernet()
        print('New kefile created at [{}]'.format(self.pathToKeyfile))
        print('Test roundtrip is good = [{}]'.format(self._testRoundTrip()))

    def encrypt(self, s):
        if self.fernetInstance == None:
            return None
        elif not type(s) == type(str()):
            print('[{}] must be a string, not [{}]'.format(s, type(s)))
            return None
        ba=s.encode('utf-8')    #convert string to bytes
        encryptedByteArray = self.fernetInstance.encrypt(ba)
        encryptedStr = codecs.decode(encryptedByteArray, InteropEncrypt.CHARACTER_SET)
        #now test to make sure decrypt will work.
        decrypted = self.decrypt(encryptedStr)
        if not decrypted == s:
            
            return None
        return encryptedStr

    def decrypt(self, s, ttlSeconds=None):
        if not ttlSeconds == None:      #convert to int if necessary
            ttlSeconds = int(ttlSeconds)
        if self.fernetInstance == None:
            return None
        elif not type(s) == type(str()):
            print('[{}] must be a string, not [{}]'.format(s, type(s)))
            return None
        ba=s.encode('utf-8')    #convert string to bytes
        try:
            decrypted = self.fernetInstance.decrypt(ba,ttlSeconds)
        except InvalidToken as ITerror:
            print('ERROR: Invalid or expired token [{}] in InteropEncrypt().decrypt() keyfile=[{}]'.format(s,self.pathToKeyfile))
            return None
        except Exception as e:
            print('ERROR: in InteropEncrypt().decrypt() for [{}] {} keyfile=[{}]'.format(s,e,self.pathToKeyfile))
            return None
        return codecs.decode(decrypted, InteropEncrypt.CHARACTER_SET)

import jwt
import json
from utils.logutils import logger

class Jwtutils:
    def __init__(self, clientname=None):
        self.logger_obj = logger()
        self.clientname = clientname
        ch = ConfigHelper(clientname=self.clientname)
        keys = ch.load_keys()
        self.client_private_key = keys[0]
        self.client_public_key = keys[1]
        self.exl_private_key = keys[2]
        self.exl_public_key = keys[3]
        self.jwtconfig = ch.getJWTSettings()

    def decode_jwt(self, token, internal = True):
        try:
            payload = jwt.decode(
                token,
                self.exl_public_key if internal else self.client_public_key,
                algorithms=[f"{self.jwtconfig['jwt_algorithm']}"],
                audience="interop.exlservice.com"
            )
            return payload
        except jwt.ExpiredSignatureError:
            self.logger_obj.log("Token expired")
            raise ValueError("Token expired")
        except jwt.InvalidTokenError:
            self.logger_obj.log("Invalid token")
            raise ValueError("Invalid token")

    def encode_jwt(self, payload):
        return jwt.encode(
            payload,
            self.exl_private_key,
            algorithm=f"{self.jwtconfig['jwt_algorithm']}"
        )

    def validate_jwt_sub(self, payload, expected_sub):
        if payload.get('sub') != expected_sub or payload.get('aud') != "interop.exlservice.com":
            self.logger_obj.log("sub claim does not match")
            raise ValueError("sub claim does not match")
        
    def validate_jwt_claims(self, payload: dict):
        """
        Validates JWT claims for 'sub', 'iss', and 'aud' against expected values.
        Returns (is_valid, mismatches) where mismatches is a list of claim names that do not match.
        Comparison is case-insensitive.

        Args:
            payload (dict): Decoded JWT payload.
            clientname (str): Expected client name for 'sub'.

        Returns:
            tuple: (bool, list) where bool is True if all claims match, False otherwise.
                list contains names of mismatched claims.
        """
        expected_iss = 'interop.exlservice.com'
        expected_aud = 'interop.exlservice.com'

        mismatches = []
        if str(payload.get('sub', '')).lower() != str(self.clientname).lower():
            mismatches.append('sub')
        if str(payload.get('iss', '')).lower() != expected_iss.lower():
            mismatches.append('iss')
        if str(payload.get('aud', '')).lower() != expected_aud.lower():
            mismatches.append('aud')
        
        return len(mismatches) == 0, mismatches

    def get_oauth_config(self):
        """
        Fetch OAuth configuration for a client from config.vw_clientmessage.
        Expected columns in view: oauthflag, oauth_api, oauth_headers, oauth_data.
        Returns a dict with safe defaults.
        """
        query = """
            SELECT
                oauthflag,
                oauth_api,
                oauth_headers,
                oauth_data
            FROM config.vw_clientmessage
            WHERE lower(client) = %s
              AND endpointdesttype = 'RESTAPI'
            LIMIT 1
        """

        defaults = {
            "oauthflag": False,
            "oauth_api": None,
            "oauth_headers": {},
            "oauth_data": {}
        }

        if not self.clientname:
            return defaults

        try:
            row = query_configdb(query, (self.clientname.lower(),))
            if not row:
                return defaults

            oauthflag = row.get("oauthflag", False)
            if isinstance(oauthflag, str):
                oauthflag = oauthflag.strip().lower() in {"1", "true", "t", "yes", "y"}

            oauth_headers = row.get("oauth_headers") or {}
            oauth_data = row.get("oauth_data") or {}

            if isinstance(oauth_headers, str):
                oauth_headers = json.loads(oauth_headers)
            if isinstance(oauth_data, str):
                oauth_data = json.loads(oauth_data)

            return {
                "oauthflag": bool(oauthflag),
                "oauth_api": row.get("oauth_api"),
                "oauth_headers": oauth_headers if isinstance(oauth_headers, dict) else {},
                "oauth_data": oauth_data if isinstance(oauth_data, dict) else {}
            }
        except Exception as e:
            self.logger_obj.log(f"Failed to fetch oauth config: {str(e)}", "warning")
            return defaults

from opentelemetry._logs import get_logger_provider, set_logger_provider

from opentelemetry.sdk._logs.export import BatchLogRecordProcessor
from opentelemetry.sdk.resources import SERVICE_NAME, Resource
from azure.monitor.opentelemetry.exporter import AzureMonitorLogExporter

deploymentplatform = (os.getenv("deploymentplatform") or 'local').lower()
logfilewithpath = os.getenv("logfilewithpath") or 'logs/interopsvc.log'

class logger:
    def __init__(self, psrvc=None):

        self.svc_name= "interopapisvc" if psrvc is None else psrvc
        # Setup file logger
        self.file_logger = logging.getLogger(self.svc_name)
        self.file_logger.setLevel(logging.INFO)

        if not any(isinstance(h, logging.FileHandler) for h in self.file_logger.handlers):
            file_handler = logging.FileHandler(logfilewithpath)
            formatter = logging.Formatter('%(asctime)s - %(levelname)s - %(message)s')
            file_handler.setFormatter(formatter)
            self.file_logger.addHandler(file_handler)
        self.file_logger.propagate = False  # Prevent duplicate logs via propagation

        # Step 2: Setup OTEL logger via LoggingHandler (only for Azure)
        self.otel_logger = None
        appinsights_conn = os.getenv('APPLICATIONINSIGHTS_CONNECTION_STRING') or ''
        if deploymentplatform == "azure" and appinsights_conn:
            # Set OTEL LoggerProvider once, with a proper Resource
            if not isinstance(get_logger_provider(), LoggerProvider):
                resource = Resource(attributes={SERVICE_NAME: self.svc_name})
                provider = LoggerProvider(resource=resource)
                set_logger_provider(provider)

                exporter = AzureMonitorLogExporter(
                    connection_string=appinsights_conn
                )
                provider.add_log_record_processor(BatchLogRecordProcessor(exporter))

            # Attach OTEL handler to Python logger
            self.otel_logger = logging.getLogger(self.svc_name)
            self.otel_logger.setLevel(logging.INFO)
            if not any(isinstance(h, LoggingHandler) for h in self.otel_logger.handlers):
                otel_handler = LoggingHandler()
                self.otel_logger.addHandler(otel_handler)
            self.otel_logger.propagate = False

    def log(self, message, msgtype=None):
        if deploymentplatform == "azure" and self.otel_logger:
            if msgtype == "error":
                self.otel_logger.error(message)
            elif msgtype == "debug":
                self.otel_logger.debug(message)
            else:
                self.otel_logger.info(message)
        else:
            if msgtype == "error":
                self.file_logger.error(message)

    def validate_claim(self):
        claim_resource = self.get_resource('claim')

        if not claim_resource:
            raise ValueError("Claim resource not found in the bundle")
        
        if claim_resource.use.lower() not in ['preauthorization', 'predetermination']:
            raise ValueError("Claim resource use must be 'preauthorization' or 'predetermination'")

        if claim_resource.status.lower() != 'active':

    def get_reference_details(self,claim_resource_update=None):
        
        if claim_resource_update is None:
            claim_resource = self.get_resource("claim")
            
        else:
            claim_resource = claim_resource_update
            
        identifiers = getattr(claim_resource, "identifier", [])
        

        if not identifiers:
            raise ValueError("No identifiers found in Claim resource.")
        if len(identifiers) > 2:
            raise ValueError("Claim contains more than 2 identifiers, which violates the PAS rule.")

        # Define priority order
        priority_order = ["official", "secondary", "usual", "temp", "old"]

        # Find identifier with highest priority use
        selected_identifier = None
        if len(identifiers) == 1:
            selected_identifier = identifiers[0]

        else:
            for priority in priority_order:
                for identifier in identifiers:
                    if getattr(identifier, "use", "") == priority:
                        selected_identifier = identifier
                        break
                if selected_identifier:
                    break

        if not selected_identifier:
            raise ValueError("No identifier with valid 'use' found. Expected one of: official, secondary, usual, temp, old.")

        value = getattr(selected_identifier, "value", "")
        system = getattr(selected_identifier, "system", "")
        if not value or not system:
            raise ValueError("Selected identifier is missing 'value' or 'system'.")

        return {
            "pasIdentifierValue": value,
            "pasIdentifierSystem": system
        }

                    condition_entry["condition"].extend(mapped_codes)

    def get_preferred_identifier(self, resource) -> str | None:
        """
        Returns the identifier value from the given FHIR resource based on identifier priority.
        If there's only one identifier, return that one; otherwise, use priority to determine the preferred identifier.

        :param resource: FHIR resource (Patient, Practitioner, etc.)
        :return: The preferred identifier value, or None if no match found
        """
        # Default priority list for identifiers
        identifier_priority_list = [
            'MB',
            'MC',
            'MA',
            'SS'
        ]

        # Check if the resource has an 'identifier' field
        if not hasattr(resource, 'identifier') or not resource.identifier:
            raise ValueError(f"Resource {resource.id} does not have an identifier field or it is empty.")

        # If there's only one identifier, return it directly without applying the priority logic
        
        identifier_list = getattr(resource, 'identifier', [])
        if len(resource.identifier) == 1:
            return resource.identifier[0].value

        # If there are multiple identifiers, apply the priority logic
        for preferred_type in identifier_priority_list:
            
                for identifier in identifier_list:
                    if hasattr(identifier, 'type') and identifier.type:
                        for coding in identifier.type.coding:
                            if coding.code.strip().lower() == preferred_type.strip().lower():
                                return identifier.value
                    else:
                        raise ValueError(f"Resource identifier type is missing or not structured as expected in {resource.id}.")

        return None  # Return None if no match is found


    def get_preferred_name(self, resource) -> str | None:
        """
        Returns the name value from the given FHIR resource based on name priority.
        If there's only one name, return that one; otherwise, use priority to determine the preferred name.

        :param resource: FHIR resource (Patient, Practitioner, etc.)
        :return: The preferred familyname,givenname value, or None if no match found
        """
        # Default priority list for names
        name_priority_list = [
            'official',  
            'maiden',
            'usual',  
            'nickname',
            'old', 
            'anonymous',
            'temporary'  
            ]

        # Check if the resource has a 'name' field
        if not hasattr(resource, 'name') or not resource.name:
            return None

        # If there's only one name, return it directly without applying the priority logic
        if len(resource.name) == 1:
            # Return the full name (given + family name)
            name = resource.name[0]
            givenname = name.given[0] if name.given else ''
            familyname = name.family if name.family else ''
            textname = name.text if name.text else ''
            prefix = name.prefix[0] if name.prefix else ''
            suffix = name.suffix[0] if name.suffix else ''
            return familyname,givenname, textname, prefix, suffix

        # If there are multiple names, apply the priority logic
        for preferred_name_type in name_priority_list:
            for name in resource.name:
                if name.use and name.use.strip().lower() == preferred_name_type.strip().lower():
                    givenname = name.given[0] if name.given else ''
                    familyname = name.family if name.family else ''
                    textname = name.text if name.text else ''
                    prefix = name.prefix[0] if name.prefix else ''
                    suffix = name.suffix[0] if name.suffix else ''
                    return familyname, givenname, textname, prefix, suffix

        return None,None,None,None,None  # Return None if no match is found

    def create_extension(self, url, value_type, value,displayrequired = False,fullurl = True, urlrequired=True):
        """
        Utility to create a FHIR Extension with a specified URL and value.
        Supports 'valueString', 'valueCodeableConcept', etc.
        """
        ext = extension.Extension()
        if urlrequired:
            if fullurl:
                ext.url = f"http://hl7.org/fhir/us/davinci-pas/StructureDefinition/extension-{url}"
            else:
                ext.url = url

        if value_type.lower() == "string":
            ext.valueString = value.get('value',"")
        elif value_type.lower() == "CodeableConcept".lower():
            # Support list of codings
            codings = []
            if displayrequired:
                if isinstance(value, list):
                    for val in value:
                        codings.append({
                            "code": val.get("code", "1"),
                            "display": val.get("display", "1")
                        })
                elif isinstance(value, dict):
                    codings = [{
                        "code": value.get("code", "1"),
                        "display": value.get("display", "1")
                    }]
            
                ext.valueCodeableConcept = codeableconcept.CodeableConcept({
                    "coding": codings
                })
            else:
                if isinstance(value, list):
                    codings = [{"code": val.get("code", "1")} for val in value]
                elif isinstance(value, dict):
                    codings = [{"code": value.get("code", "1")}]

                ext.valueCodeableConcept = codeableconcept.CodeableConcept({
                    "coding": codings
                })

        
        elif value_type.lower() == "reference":
            # Expecting a dict with at least a 'reference' key
            reference = fhirreference.FHIRReference()
            reference.reference = value.get("value", "")
            ext.valueReference = reference
        elif value_type.lower() == "quantity":
            ext.valueQuantity = quantity_.Quantity({
                                    "value": value.get("value", 0.0),
                                    "code": value.get("code", "")
                                })
            
        elif value_type.lower() == "date":
            ext.valueDate = fhirdate.FHIRDate(value)

        elif value_type.lower() == "period":
            p = period.Period()
            p.start = fhirdatetime.FHIRDateTime(value.get("start")) if value.get("start") else None
            p.end = fhirdatetime.FHIRDateTime(value.get("end")) if value.get("end") else None
            ext.valuePeriod = p
        else:
            raise ValueError(f"Unsupported value_type: {value_type}")

        return ext

    def get_extension_value(self,extensions, filter_key,child_filter=None):
        """
        Searches FHIR extensions for a URL containing the filter_key and returns its value.

        Args:
            extensions (list): List of FHIR Extension objects.
            filter_key (str): Keyword to search for in extension URLs.

        Returns:
            Any: The value of the matching extension, or None if not found.
        """
        if not extensions or not isinstance(extensions, list):
            self.logger_obj.log("No extensions found or invalid type provided.")
            return None
        
        for ext in extensions:
            if not ext.url or filter_key.lower() not in ext.url.lower():
                continue
            self.logger_obj.log(f"Found extension with URL: {ext.url}")

            # Check and return appropriate value type
            if hasattr(ext, "valueDate") and ext.valueDate:
                return ext.valueDate.isostring
            elif hasattr(ext, "valueString") and ext.valueString:
                return ext.valueString
            elif hasattr(ext, "valueBoolean") and ext.valueBoolean is not None:
                return ext.valueBoolean
            elif hasattr(ext, "valueInteger") and ext.valueInteger is not None:
                return ext.valueInteger
            elif hasattr(ext,"valueDecimal") and ext.valueDecimal is not None:
                return ext.valueDecimal
            elif child_filter is not None and  hasattr(ext,"extension") and ext.extension:
                self.logger_obj.log(f"Searching for child filter: {child_filter} in extension: {ext.url}")
                return self.get_extension_value(ext.extension, child_filter)
            elif hasattr(ext, "valueCodeableConcept") and ext.valueCodeableConcept:
                #if ext.valueCodeableConcept.coding:
                #    return {
                #        "code": ext.valueCodeableConcept.coding[0].code,
                #        "display": ext.valueCodeableConcept.coding[0].display
                #    }
                return self.get_codeableconcept_codes(ext.valueCodeableConcept)
            elif hasattr(ext, "valuePeriod") and ext.valuePeriod is not None:
                return {
                    "start": ext.valuePeriod.start.date.date().isoformat() if ext.valuePeriod.start else "",
                    "end": ext.valuePeriod.end.date.date().isoformat() if ext.valuePeriod.end else ""
                }
            elif hasattr(ext, "valueCode") and ext.valueCode is not None:
                return ext.valueCode
            elif hasattr(ext, "timingDate") and ext.timingDate:
                return ext.timingDate.isostring
            elif hasattr(ext, "timingPeriod") and ext.timingPeriod is not None:
                return {
                    "start": ext.timingPeriod.start.date.date().isoformat() if ext.timingPeriod.start else "",
                    "end": ext.timingPeriod.end.date.date().isoformat() if ext.timingPeriod.end else ""
                }

        return None  # If no match is found
    
    def get_codeableconcept_codes(self, codeable_concept):
        """
        Extracts all 'code' and 'display' values from a CodeableConcept.coding list.
        Returns a list of dicts: [{'code': ..., 'display': ...}, ...]
        """
        if not codeable_concept:
            return None#[{"code": "", "display": ""}]

        codings = getattr(codeable_concept, "coding", None) or codeable_concept.get("coding", [])
        if not codings:
            return None
        
        results = []
        for coding in codings:
            code = coding.code or "" #getattr(coding, "code", None) or coding.get("code", None)
            display = coding.display or "" #getattr(coding, "display", None) or coding.get("display", None)
            if code or display:
                results.append({
                    "code": code or "",
                    "display": display or ""
                })
        return results if results else [{"code": "", "display": ""}]


    def dict_to_codeableconcept(self,data):
        """
        Converts a dict with keys 'code' and 'description' to a CodeableConcept.
        Returns None if both code and description are missing/empty.
        """
        code = data.get("code", "")
        description = data.get("description", "")
        if not code and not description:
            return None

        cc = codeableconcept.CodeableConcept()
        coding = c1.Coding()
        coding.code = code
        coding.display = description
        cc.coding = [coding]
        return cc



    def generated_pas_sequence_for_new_service(self, data_dict, external_pas_map, claimtype):
        """
        Updates pasSequence for services in data_dict using external_pas_map.
        If pasSequence == '-1', generates a new one and appends to external_pas_map.
        If claimtype == 'claim' and external_pas_map is an empty dict, initialize it from servicePASMapping and return early.

        Parameters:
            data_dict (dict): Input dictionary containing authorization.services
            external_pas_map (list or dict): crSequence to pasSequence mappings
            claimtype (str): Type of claim, affects processing flow

        Returns:
            tuple:
                dict: Updated data_dict
                list: Updated external_pas_map (with generated sequences if any)
        """
        auth = data_dict.get('authorization', {}) or {}
        services = auth.get('services', []) or []

        if not isinstance(services, list) or not services:
            self.logger_obj.log("No new services found in the authorization section.", 'info')
            return data_dict, external_pas_map

        # ? Step 1: Special handling for claimtype == 'claim' and empty dict
        if claimtype.lower() == 'claim':
            self.logger_obj.log("Claim type is 'claim' and external_pas_map is empty. Initializing from services.", 'info')
            for svc in services:
                mapping = svc.get("servicePASMapping", {})
                cr_seq = str(mapping.get("crSequence", "")).strip()
                pas_seq = str(mapping.get("pasSequence", "")).strip()
                if cr_seq and pas_seq:
                    external_pas_map.append({
                        "crSequence": cr_seq,
                        "pasSequence": pas_seq
                    })
            # ? Skip all further processing for 'claim'
            return data_dict, external_pas_map

        # Convert to dict for fast lookup: crSequence -> pasSequence
        external_pas_map_dict = {
            str(item.get("crSequence")).strip(): item.get("pasSequence")
            for item in external_pas_map
            if "crSequence" in item and "pasSequence" in item
        }

        generated_any = False
        updated_external_pas_map = copy.deepcopy(external_pas_map)
        self.logger_obj.log(f"after deepcopy: {updated_external_pas_map}")

        # Assign pasSequence from external_pas_map
        for item in services:
            mapping = item.get("servicePASMapping", {}) or {}
            cr_seq = str(mapping.get("crSequence", "")).strip()

            if cr_seq and cr_seq in external_pas_map_dict:
                mapping["pasSequence"] = external_pas_map_dict[cr_seq]

            item["servicePASMapping"] = mapping

        # Find max numeric pasSequence
        existing_pas = [
            int(item.get("pasSequence"))
            for item in external_pas_map
            if str(item.get("pasSequence", "")).isdigit()
        ]

        next_pas = max(existing_pas, default=0) + 1

        self.logger_obj.log(f"Next available PAS sequence to assign: {next_pas}")

        # Generate new PAS sequences where needed
        for i in services:
            mapping = i.get("servicePASMapping", {}) or {}
            cr_seq = str(mapping.get("crSequence", "")).strip()
            pas_seq_cur = str(mapping.get("pasSequence", "")).strip()

            if pas_seq_cur == "-1" and cr_seq:
                mapping["generatedPasSequence"] = str(next_pas)

                updated_external_pas_map.append({
                    "crSequence": cr_seq,
                    "pasSequence": str(next_pas)
                })
                generated_any = True
                next_pas += 1

        self.logger_obj.log(f"Updated services with generated PAS sequences: {services}")
        
        self.logger_obj.log(f"Updated external PAS map: {updated_external_pas_map}")

        return data_dict, updated_external_pas_map if generated_any else external_pas_map

    def extract_telecom_fields(self,resource):
        telecoms = {
            "homePhone": "",
            "workPhone": "",
            "mobilePhone": "",
            "email": "",
            "fax": ""
        }

        if not hasattr(resource, 'telecom') or not resource.telecom:
            return telecoms

        # Group telecoms by (system, use)
        

        grouped = defaultdict(list)
        for t in resource.telecom:
            system = str(getattr(t, "system", "") or "").lower()
            use = str(getattr(t, "use", "") or "").lower()
            if system and getattr(t, 'value', None):
                grouped[(system, use)].append(t)

        # For each group, apply ranking only if multiple entries exist
        for (system, use), items in grouped.items():
            if len(items) > 1:
                # pick the one with the lowest rank (e.g., rank == 1)
                items = sorted(items, key=lambda x: getattr(x, 'rank', 9999))
                selected = items[0]
            else:
                selected = items[0]

            value = getattr(selected, 'value', None)
            if not value:
                continue

            # Map to telecoms dict
            if system == 'phone':
                if use == 'home':
                    telecoms['homePhone'] = value
                elif use == 'work':
                    telecoms['workPhone'] = value
                elif use == 'mobile':
                    telecoms['mobilePhone'] = value
            elif system == 'email':
                telecoms['email'] = value
            elif system == 'fax':
                telecoms['fax'] = value

        return telecoms
    
    def build_telecom_from_contact_details(self, contact_details: dict):

    def get_identifier(self,resource):

        providerid, providernpi, providertin, providerid_system = '', '', '', ''
        
        # Check if the resource has an 'identifier' field
        if not hasattr(resource, 'identifier') or not resource.identifier:
            return providerid, providernpi, providertin, providerid_system

        # If there's only one identifier, return it directly without applying the priority logic
        #if len(resource.identifier) == 1:
        #    providerid = resource.identifier[0].value
        #    return providerid, providernpi, providertin

        # NPI/TIN are identified using Identifier.system fixed values.
        npi_system_values = {"http://hl7.org/fhir/sid/us-npi", "us-npi"}
        tin_system_values = {"http://terminology.hl7.org/namingsystem/usein", "usein"}

        identifiers = getattr(resource, 'identifier', []) or []

        # Loop 1: providerid should only come from type code PRN when type is available.
        for identifier in identifiers:
            identifier_value = getattr(identifier, 'value', '') or ''
            providerid_system = getattr(identifier, 'system', '') or ''
            identifier_type = getattr(identifier, 'type', None) or None
            codings = getattr(identifier_type, 'coding', []) or [] if identifier_type else []

            for coding in codings:
                code_value = (getattr(coding, 'code', '') or '').strip().lower()
                if code_value == "prn":
                    providerid = identifier_value
                    break

        # Loop 2: fetch NPI and TIN from Identifier.system fixed values.
        for identifier in identifiers:
            identifier_value = getattr(identifier, 'value', '') or ''
            system_value = (getattr(identifier, 'system', '') or '').strip().lower()
            system_tail = system_value.rstrip('/').split('/')[-1] if system_value else ''

            if (system_value in npi_system_values or system_tail in npi_system_values) and identifier_value:
                providernpi = identifier_value
            if (system_value in tin_system_values or system_tail in tin_system_values) and identifier_value:
                providertin = identifier_value
        
        return providerid, providernpi, providertin, providerid_system

            error.code = codeableconcept.CodeableConcept()
            error.code.coding = [c1.Coding({
                "system": "http://terminology.hl7.org/CodeSystem/adjudication-reason",
                "code": reject.get("code", ""),
                "display": reject.get("display", "")
            })]

        # Follow-up action ? extension
        followup = error_data.get("followUpAction", {})
        if followup:
            followup_cc = codeableconcept.CodeableConcept()
            followup_cc.coding = [c1.Coding({
                "system": "http://terminology.hl7.org/CodeSystem/followup-action",
                "code": followup.get("code", ""),
                "display": followup.get("display", "")
            })]

            followup_ext = extension.Extension()
            followup_ext.url = "http://example.org/fhir/StructureDefinition/errorFollowUpAction"
            followup_ext.valueCodeableConcept = followup_cc
            error.extension = [followup_ext]

        cr.error = [error]
        return cr.as_json()

    def remove_supporting_info(self, bundle_json):
        """

        Case-insensitive match on 'resourceType'.
        """
        # Make a deep copy to avoid modifying the original input
        cleaned_bundle = copy.deepcopy(bundle_json)

        for entry in cleaned_bundle.get("entry", []):
            resource = entry.get("resource", {})
            if resource.get("resourceType", "").lower() == "documentreference":
                for content_item in resource.get("content", []):
                    attachment = content_item.get("attachment", {})
                    if isinstance(attachment, dict):
                        attachment.pop("data", None)

    def upload_attachments_to_storage(self, base64_data, unique_file_id, container_name):
        """
        Upload base64 encoded data to Azure Blob Storage using chunked uploads for large files.
        
        Args:
            base64_data (str): Base64 encoded file data
            unique_file_id (str): Unique identifier for the file
            container_name (str): Azure blob container name
            chunk_size (int): Size of each chunk in bytes (default 4MB)
        
        Returns:
            str: Blob URL if successful, None if failed
        """
        try:
            def calculate_chunk_size(file_size):
                """Dynamically calculate chunk size based on file size."""
                if file_size < 10 * 1024 * 1024:  # less than 10 MB
                    return 2 * 1024 * 1024  # 2 MB
                elif file_size < 100 * 1024 * 1024:  # 10 MB to 100 MB
                    return 10 * 1024 * 1024  # 10 MB
                else:  # Larger than 100 MB
                    return 20 * 1024 * 1024  # 20 MB
            
            
            # Azure Storage Connection String
            AZURE_STORAGE_CONNECTION_STRING = os.getenv("BLOB_STORAGE_CONNECTION_STRING")

            if not AZURE_STORAGE_CONNECTION_STRING:
                return "Azure Storage connection string is missing", 500
            
            # Create blob service client
            blob_service_client = BlobServiceClient.from_connection_string(AZURE_STORAGE_CONNECTION_STRING)
            blob_name = f"{unique_file_id}"
            blob_client = blob_service_client.get_blob_client(container=container_name, blob=blob_name)
            
            # Convert string to bytes
            data_bytes = base64_data.encode('utf-8')
            data_size = len(data_bytes)
            chunk_size = calculate_chunk_size(data_size)
            
            # If file is small, use regular upload
            if data_size <= chunk_size:
                data_stream = BytesIO(data_bytes)
                blob_client.upload_blob(data_stream, overwrite=True, content_type="text/plain")
                return blob_client.url
            
            # For large files, use chunked upload
            self.logger_obj.log(f"Uploading large file ({data_size} bytes) in chunks of {chunk_size} bytes")
            
            # Stage blocks for chunked upload
            block_list = []
            data_stream = BytesIO(data_bytes)
            
            block_number = 0
            while True:
                chunk = data_stream.read(chunk_size)
                if not chunk:
                    break
                    
                # Generate block ID (must be base64 encoded and same length for all blocks)
                block_id = base64.b64encode(f"{block_number:08d}".encode()).decode()
                
                # Stage the block
                blob_client.stage_block(block_id, chunk)
                block_list.append(block_id)
                block_number += 1
                
                # Log progress
                progress = min(100, (block_number * chunk_size / data_size) * 100)
                self.logger_obj.log(f"Upload progress: {progress:.1f}% (Block {block_number})")
            
            # Commit all blocks to create the blob
            blob_client.commit_block_list(block_list)
            self.logger_obj.log(f"Successfully uploaded file with {len(block_list)} blocks")
            
            return blob_client.url
            
        except Exception as e:
            self.logger_obj.log(f"Error uploading file to blob storage: {str(e)}", "error")
            return None

def insert_tracking_record(tracking_data: dict):
    """
    Inserts a record into config.tracking using the provided dictionary.

    tracking_data["Success"] = "Success" if tracking_data.get("SuccessFlag") == 1 else "Failed"
    tracking_data["InsertDTS"] = tracking_data.get("InsertDTS") or datetime.now(timezone.utc)
    tracking_data["InsertUser"] = tracking_data.get("InsertUser") or "Interopservice"

    try:
        tracking_collection.insert_one(tracking_data)
    except Exception as e:
        raise e
