CREATE OR REPLACE TABLE gold_vbc_insights AS SELECT *, CASE WHEN Composite_VBC_Score < 0.55 AND Preventive_Score < 0.5 THEN 'Low composite score driven by poor preventive gap closure' WHEN Composite_VBC_Score < 0.55 AND Chronic_Score < 0.5 THEN 'Low composite score due to weak chronic care control' WHEN Composite_VBC_Score < 0.55 AND Care_Coordination_Score < 0.5 THEN 'Low VBC score due to poor TRC and PPC follow-up' WHEN Composite_VBC_Score < 0.55 AND GSD_Closure < 0.5 THEN 'Low score due to low GSD behavioral health gap closure' ELSE 'Provider performing within expected range' END AS Provider_Insight FROM gold_vbc_final;
