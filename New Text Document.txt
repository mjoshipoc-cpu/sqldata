SELECT
  Provider_ID,
  Year,
  Measure,
  Gap_Closure
FROM gold_vbc_domain
LATERAL VIEW stack(
  8,                          -- number of measures
  'BPD', BPD_Closure,
  'CBP', CBP_Closure,
  'COA', COA_Closure,
  'GSD', GSD_Closure,
  'LSC', LSC_Closure,
  'PPC', PPC_Closure,
  'TRC', TRC_Closure,
  'WCC', WCC_Closure
) s(Measure, Gap_Closure);
