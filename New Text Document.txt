"""
python tests/test_pipeline_real_db.py testclient Observation
python tests/test_pipeline_real_db.py testclient ServiceRequest
python tests/test_pipeline_real_db.py testclient DiagnosticReport
python tests/test_pipeline_real_db.py testclient DocumentReference
python tests/test_pipeline_real_db.py testclient ImagingStudy

Full pipeline flow test against your REAL configured MongoDB, for any of the
5 resource types the pipeline supports (Observation, DiagnosticReport,
ServiceRequest, DocumentReference, ImagingStudy).

Runs the actual pipeline code - ingest (parse/validate/canonicalize),
persist_single (create/update/delete), and search.py (read/history) - against
whatever Mongo is configured in interop.config's [repodb] section. No Flask,
no JWT/HTTP layer.

Requires interop.config's [repodb] filled in with real Mongo details, and
[msgqueue] to have placeholder username/password (config loader needs them
regardless of what you're testing).

Usage:
    python tests/test_pipeline_real_db.py <clientname> [resource_type]

    e.g. python tests/test_pipeline_real_db.py testclient
         python tests/test_pipeline_real_db.py testclient ServiceRequest

Exit code 0 = all good, 1 = something failed.
"""
import os
import sys
import json

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(HERE)
SRC_DIR = os.path.join(ROOT, "src", "interop")
sys.path.insert(0, SRC_DIR)

from utils.db_utils import get_mongo_db_conn  # noqa: E402
from utils.logutils import logger  # noqa: E402
from fhirpipeline.pipeline import ingest  # noqa: E402
from fhirpipeline.persist import persist_single  # noqa: E402
from fhirpipeline.search import read_by_id, read_history  # noqa: E402

if len(sys.argv) < 2:
    sys.exit("Usage: python tests/test_pipeline_real_db.py <clientname> [resource_type]")

clientname = sys.argv[1]
resource_type = sys.argv[2] if len(sys.argv) > 2 else "Observation"
logger_obj = logger()


def get_collection(clientname, collection_name):
    return get_mongo_db_conn(clientname, collection_name, logger_obj)


_p = _f = 0
def check(label, cond):
    global _p, _f
    if cond:
        _p += 1; print(f"  ok   {label}")
    else:
        _f += 1; print(f"  FAIL {label}")


# --------------------------------------------------------------------------
# one valid sample + one bad-status variant per supported resource type
# --------------------------------------------------------------------------
SAMPLES = {
    "Observation": {
        "resourceType": "Observation", "status": "final",
        "code": {"coding": [{"system": "http://loinc.org", "code": "8310-5"}]},
        "subject": {"reference": "Patient/123"},
        "effectiveDateTime": "2026-09-09T10:00:00Z",
        "valueQuantity": {"value": 37.1, "unit": "Cel",
                          "system": "http://unitsofmeasure.org", "code": "Cel"},
    },
    "DiagnosticReport": {
        "resourceType": "DiagnosticReport", "status": "final",
        "code": {"coding": [{"system": "http://loinc.org", "code": "58410-2"}]},
        "subject": {"reference": "Patient/123"},
    },
    "ServiceRequest": {
        "resourceType": "ServiceRequest", "status": "active", "intent": "order",
        "subject": {"reference": "Patient/123"},
        "code": {"coding": [{"system": "http://loinc.org", "code": "58410-2"}]},
    },
    "DocumentReference": {
        "resourceType": "DocumentReference", "status": "current",
        "content": [{"attachment": {"contentType": "application/pdf",
                                     "url": "Binary/example"}}],
    },
    "ImagingStudy": {
        "resourceType": "ImagingStudy", "status": "available",
        "subject": {"reference": "Patient/123"},
    },
}
BAD_STATUS = {
    "Observation": "bogus-not-a-real-status",
    "DiagnosticReport": "bogus-not-a-real-status",
    "ServiceRequest": "bogus-not-a-real-status",
    "DocumentReference": "bogus-not-a-real-status",
    "ImagingStudy": "bogus-not-a-real-status",
}

if resource_type not in SAMPLES:
    sys.exit(f"Unknown resource_type '{resource_type}'. Choose from: {', '.join(SAMPLES)}")

RES = SAMPLES[resource_type]

print(f"=== 1. ingest + create ({resource_type}, client='{clientname}') ===")
result = ingest(json.dumps(RES).encode(), "application/fhir+json", resource_type)
check("ingest ok", result.ok)
if not result.ok:
    print(json.dumps(result.outcome, indent=2))
    sys.exit(1)

pr = persist_single(get_collection, clientname, result.canonical, "create")
check("persist create ok", pr.ok)
check("status 201", pr.status == 201)
oid = (pr.resource or {}).get("id")
check("server-assigned id", bool(oid))
check("versionId == '1'", (pr.resource or {}).get("meta", {}).get("versionId") == "1")
if not pr.ok or not oid:
    print(json.dumps(pr.outcome, indent=2))
    sys.exit(1)
print(f"  -> {resource_type}/{oid} written to real DB")

print("\n=== 2. read back ===")
status, body, etag = read_by_id(get_collection, clientname, resource_type, oid)
check("200 OK", status == 200)
check("same id", (body or {}).get("id") == oid)
check("ETag W/\"1\"", etag == 'W/"1"')

print("\n=== 3. update -> v2 ===")
result2 = ingest(json.dumps({**RES, "status": pr.resource["status"]}).encode(),
                  "application/fhir+json", resource_type)
check("ingest (update) ok", result2.ok)
pr2 = persist_single(get_collection, clientname, result2.canonical, "update", resource_id=oid)
check("persist update ok", pr2.ok)
check("versionId == '2'", (pr2.resource or {}).get("meta", {}).get("versionId") == "2")

print("\n=== 4. history ===")
hstatus, hbody = read_history(get_collection, clientname, resource_type, oid)
check("200 + type history", hstatus == 200 and hbody.get("type") == "history")
check("2 versions", len(hbody.get("entry", [])) == 2)

print("\n=== 5. delete ===")
pr3 = persist_single(get_collection, clientname, {"resourceType": resource_type},
                     "delete", resource_id=oid)
check("204 No Content", pr3.status == 204)

status4, body4, _ = read_by_id(get_collection, clientname, resource_type, oid)
check("read after delete -> 404", status4 == 404)

print("\n=== 6. validation actually runs (bad data must be rejected, nothing written) ===")
bad = {**RES, "status": BAD_STATUS[resource_type]}
bad_result = ingest(json.dumps(bad).encode(), "application/fhir+json", resource_type)
check("invalid status -> ingest rejects it", not bad_result.ok)
check("422 Unprocessable", bad_result.status == 422)
if bad_result.outcome:
    print("  OperationOutcome:", json.dumps(bad_result.outcome, indent=2))

print(f"\n{_p} passed, {_f} failed")
sys.exit(1 if _f else 0)
