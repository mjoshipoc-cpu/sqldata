CREATE OR REPLACE TABLE gold_vbc_composite AS
SELECT *,
0.32*Chronic_Score +
0.28*Preventive_Score +
0.20*Care_Coordination_Score +
0.20*Behavioral_Score AS Composite_VBC_Score
FROM gold_vbc_domain;


CREATE OR REPLACE TABLE gold_vbc_final AS
SELECT *,
CASE
  WHEN NTILE(5) OVER (ORDER BY Composite_VBC_Score DESC) = 1 THEN 'Gold'
  WHEN NTILE(5) OVER (ORDER BY Composite_VBC_Score DESC) IN (2,3) THEN 'Silver'
  ELSE 'Bronze'
END AS Provider_Tier
FROM gold_vbc_composite;
