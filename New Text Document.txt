SELECT
  SUM(
      CASE WHEN BPD_Gap = 'N' THEN 1 ELSE 0 END
    + CASE WHEN CBP_Gap = 'N' THEN 1 ELSE 0 END
    + CASE WHEN COA_Gap = 'N' THEN 1 ELSE 0 END
    + CASE WHEN GSD_Gap = 'N' THEN 1 ELSE 0 END
    + CASE WHEN LSC_Gap = 'N' THEN 1 ELSE 0 END
    + CASE WHEN PPC_Gap = 'N' THEN 1 ELSE 0 END
    + CASE WHEN TRC_Gap = 'N' THEN 1 ELSE 0 END
    + CASE WHEN WCC_Gap = 'N' THEN 1 ELSE 0 END
  ) * 1.0
  /
  SUM(
      CASE WHEN BPD_Gap IN ('Y','N') THEN 1 ELSE 0 END
    + CASE WHEN CBP_Gap IN ('Y','N') THEN 1 ELSE 0 END
    + CASE WHEN COA_Gap IN ('Y','N') THEN 1 ELSE 0 END
    + CASE WHEN GSD_Gap IN ('Y','N') THEN 1 ELSE 0 END
    + CASE WHEN LSC_Gap IN ('Y','N') THEN 1 ELSE 0 END
    + CASE WHEN PPC_Gap IN ('Y','N') THEN 1 ELSE 0 END
    + CASE WHEN TRC_Gap IN ('Y','N') THEN 1 ELSE 0 END
    + CASE WHEN WCC_Gap IN ('Y','N') THEN 1 ELSE 0 END
  ) AS Overall_Gap_Closure
FROM silver_provider_clean;
