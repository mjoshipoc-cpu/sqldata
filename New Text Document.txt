CREATE OR REPLACE TABLE gold_gap_closure AS
SELECT
  Provider_ID,
  COUNT(*) AS Members,
  AVG(BP_Controlled) AS BP_Control_Rate,
  AVG(A1c_Controlled) AS A1c_Control_Rate,
  AVG(LDL_Controlled) AS LDL_Control_Rate,

  SUM(CASE WHEN BPD_Gap = 'N' THEN 1 END) * 1.0 /
  SUM(CASE WHEN BPD_Gap IN ('Y','N') THEN 1 END) AS BPD_Closure,

  SUM(CASE WHEN CBP_Gap = 'N' THEN 1 END) /
  SUM(CASE WHEN CBP_Gap IN ('Y','N') THEN 1 END) AS CBP_Closure,

  SUM(CASE WHEN COA_Gap='N' THEN 1 END) /
  SUM(CASE WHEN COA_Gap IN ('Y','N') THEN 1 END) AS COA_Closure,

  SUM(CASE WHEN GSD_Gap='N' THEN 1 END) /
  SUM(CASE WHEN GSD_Gap IN ('Y','N') THEN 1 END) AS GSD_Closure,

  SUM(CASE WHEN LSC_Gap='N' THEN 1 END) /
  SUM(CASE WHEN LSC_Gap IN ('Y','N') THEN 1 END) AS LSC_Closure,

  SUM(CASE WHEN PPC_Gap='N' THEN 1 END) /
  SUM(CASE WHEN PPC_Gap IN ('Y','N') THEN 1 END) AS PPC_Closure,

  SUM(CASE WHEN TRC_Gap='N' THEN 1 END) /
  SUM(CASE WHEN TRC_Gap IN ('Y','N') THEN 1 END) AS TRC_Closure,

  SUM(CASE WHEN WCC_Gap='N' THEN 1 END) /
  SUM(CASE WHEN WCC_Gap IN ('Y','N') THEN 1 END) AS WCC_Closure,

  AVG(PMPM_Cost) AS Avg_PMPM,
  SUM(ER_Visits) / COUNT(*) * 1000 AS ER_Visits_per_1000,
  SUM(Hospital_Admits) / COUNT(*) * 1000 AS Admit_Rate_per_1000,
  AVG(Readmission_30_Day) AS Readmission_Rate

FROM gold_clinical
GROUP BY Provider_ID;
