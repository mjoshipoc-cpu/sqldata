CREATE OR REPLACE TABLE gold_vbc_domain AS
SELECT *,
(BP_Control_Rate + A1c_Control_Rate + LDL_Control_Rate) / 3 AS Chronic_Score,

(BPD_Closure + COA_Closure + LSC_Closure + WCC_Closure) / 4 AS Preventive_Score,

(PPC_Closure + TRC_Closure) / 2 AS Care_Coordination_Score,

GSD_Closure AS Behavioral_Score

FROM gold_gap_closure;
