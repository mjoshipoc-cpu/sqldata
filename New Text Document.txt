"""
Insert one sample Observation resource into the 'observation' collection
using the project's real config (interop.config -> ConfigHelper -> db_utils),
the same connection path the app itself uses. No Flask, no auth, no pipeline -
just proves the configured Mongo connection can write/read.

Requires interop.config's [repodb] section to be filled in with real
hostname/port/database/username/password (password must be encrypted -
see utils/InteropEncrypt.py).

Usage:
    python tests/insert_observation.py <clientname>

    e.g. python tests/insert_observation.py testclient
"""
import os
import sys
from datetime import datetime, timezone

HERE = os.path.dirname(os.path.abspath(__file__))
ROOT = os.path.dirname(HERE)
SRC_DIR = os.path.join(ROOT, "src", "interop")
sys.path.insert(0, SRC_DIR)

from utils.db_utils import get_mongo_db_conn  # noqa: E402
from utils.logutils import logger  # noqa: E402

clientname = sys.argv[1]
logger_obj = logger()

print(f"Connecting for client '{clientname}'...", flush=True)
collection = get_mongo_db_conn(clientname, "observation", logger_obj)
print("Connected. Collection handle obtained.", flush=True)

sample_observation = {
    "resourceType": "Observation",
    "status": "final",
    "code": {
        "coding": [{"system": "http://loinc.org", "code": "8310-5", "display": "Body temperature"}]
    },
    "subject": {"reference": "Patient/123"},
    "effectiveDateTime": datetime.now(timezone.utc).isoformat(),
    "valueQuantity": {"value": 37.1, "unit": "Cel", "system": "http://unitsofmeasure.org", "code": "Cel"},
    "meta": {"versionId": "1"},
}

print("Inserting...", flush=True)
result = collection.insert_one(sample_observation)
print("Inserted _id:", result.inserted_id, flush=True)

doc = collection.find_one({"_id": result.inserted_id})
print("Read back:", doc, flush=True)
