from pyspark.sql.functions import rand, when, floor, lit, expr

df = spark.table("silver_provider_clean")

df_random = df.withColumn(
    "Random_Year",
    when(rand() < 0.5, lit(2024)).otherwise(lit(2025))
).withColumn(
    "Random_Month",
    (floor(rand()*12) + 1).cast("int")
).withColumn(
    "Random_Day",
    (floor(rand()*28) + 1).cast("int")
).withColumn(
    "Random_Date",
    expr("to_date(concat(Random_Year,'-',Random_Month,'-',Random_Day))")
)
